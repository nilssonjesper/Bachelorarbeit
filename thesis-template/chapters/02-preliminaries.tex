\begin{definition}[Convergence of Geometric Series]
    The geometric series $\sum_{k=0}^\infty q^k$ converges for all $q\in\RR$ with $|q|<1$, where
    \[\sum_{k=0}^\infty q^k=\frac{1}{1-q}\ ,\ \text{ for } |q|<1\]
\end{definition}

\begin{definition}[Gershgorin circle theorem]\label{def:gershgorin_theorem}
    Let $\tT\in \CC^{n\times n}$ be a square complex matrix, then for every eigenvalue $\lambda$ of $\tT$ it is true that 
    \[|\lambda-(\tT)_{ii}|\leq\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|\ ,\] 
    for any $i\in \underline{n}$ and where $|\cdot|$ denotes the absolute value, i.e. magnitude.\\
    In other words, all eigenvalues of $\tT$ lie on a disk in the complex plane centered at $(\tT)_{ii}$ with radius $\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|$.
\end{definition}
Matrix invertible if 1 not eigenvalue

\begin{definition}[Row-Stochastic Matrix]\label{def:row-stoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-stochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} = 1$ for all rows $i \in \underline{n}$.
\end{definition}
When considering each row as a function $f_i(j)=(\tT)_{ij}$, these functions are discrete probability distributions.

\begin{definition}[(Strictly) Row-Substochastic Matrix]\label{def:row-substoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-substochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} \leq 1$ for all rows $i \in \underline{n}$.\\
    $\tT$ is strictly row-substochastic if and only if ther exists at least one row $i'$ with $\sum_{j=0}^n(\tT)_{i'j} < 1$
\end{definition}

\begin{theorem}\label{th:row-substoch_eigenvalue_0}
    If a matrix $\tT$ is strictly row-substochastic, then it has no eigenvalue $|\lambda| \geq 1$.
\end{theorem}

\begin{proof}
    Let us assume that $|\lambda| > 1$ is an eigenvalue of $\tT$. As $(\tT)_{ij}\in \RR_{\geq 0}$, using \cref{def:gershgorin_theorem}
    \[
    |\lambda-(\tT)_{ii}| = |\lambda|-(\tT)_{ii} \leq \sum_{j=1\land j\neq i}^n|(\tT)_{ij}|\]
    \[ \iff 1 < |\lambda| \leq (\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij}\ .
    \]
    \todo{weil $\tT$ non negative}
    However, as $\tT$ is  strictly row-substochastic, we know that $(\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij} < 1 < |\lambda|$ for all $i \in \underline{n}$.
    Thus, any strictly row-substochastic matrix $\tT$ has no eigenvalues $|\lambda| > 1$.
\end{proof}

\begin{definition}[Jordan normal form]\label{def:jordan_normal_form}

    Every square matrix $A \in \CC^{n\times n}$ is similar to a block diagonal matrix $J \in \CC^{n\times n}$, also called the Jordan normal form of $A$, meaning $A = P J P^{-1}$ for an invertible matrix $P \in \CC^{n\times n}$. The blocks on the diagonal of the Jordan matrix are called Jordan blocks $J_i$ and have the form
\[J_i = \begin{bmatrix}
\lambda_i & 1 & 0 & \dots & 0 \\
0 & \lambda_i & \ddots  & \ddots & \vdots \\
\vdots & \ddots & \ddots & 1 & 0 \\
0 & \ddots & 0 & \lambda_i & 1 \\
0 & 0 & \dots & 0 & \lambda_i
\end{bmatrix}\ , \]
where $\lambda_i$ is the $i$-th eigenvalue of $A$. 
\end{definition}

\begin{definition}[Matrix inversion by the Neumann Series]\label{def:neumann_series}
    Given a matrix $\tT\in \RR^{n\times n}$ with $\lim_{k\rightarrow \infty}\tT^k = 0$, i.e. $\tT$ is convergent, the Neumann series $\sum_{k=0}^\infty \tT^k$ converges against $(\eye - \tT)^{-1}$.
\end{definition}

\begin{definition}[Irreducible and Reducible Matrix]
    A non-negative square matrix $\tT\in \RR_{\geq 0}^{n\times n}>0$ with $n\geq 2$ is irreducible if and only if $\sum_{k=0}^{n-1}\tT^k>0$. If $\tT$ is not irreducible, it is reducible.
\end{definition}

\begin{definition}[Permutation Matrix]
    A binary square matrix $P\in\{0,1\}^{n\times n}$ is a permutation matrix if and only if
    \begin{enumerate}[label=(\roman*)]
        \item $(P)_{ij} = 1 = (P)_{i'j}$ with $i\neq i'$       \item $(P)_{ij} = 1 = (P)_{ij'}$ with $j\neq j'$
    \end{enumerate}
    ($P^{-1} = P^\T$)
\end{definition}
In other words exactly one entry of $P$ is 1 per row and column.

\begin{definition}[Upper and Lower Triangular Matrix]
    An upper/lower triangular matrix is a square matrix $\tT_{up},\tT_{lo}\in \RR^{n\times n}$, where all entries below/above the diagonal are 0.
    \[\tT_{up}=\begin{bmatrix}
        a_{11} & \dots & a_{1n} \\
         & \ddots & \vdots \\
        0 & & a_{nn}
    \end{bmatrix}
    \ \ 
    \tT_{lo}=\begin{bmatrix}
        a_{11} &  & 0\\
        \vdots & \ddots &  \\
        a_{n1} & \dots & a_{nn}
    \end{bmatrix}
    \]
\end{definition}

\begin{definition}[Discrete Probability Distribution]
    A function $P$ is a \textit{discrete probability distribution}, if $P: X \rightarrow [0,1]$ where X is a countable set and $\sum_{x\in X}P(x) = 1$.
\end{definition}

\begin{definition}[Conditional Probability]
    $Pr(A \mid B) = \frac{Pr(A\cap B)}{Pr(B)}$
\end{definition}
