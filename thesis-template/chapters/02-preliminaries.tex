\section{Notation}\label{sec:notation}
We denote the set of natural numbers, including zero, as $\NN$ and the set of positive natural numbers, excluding zero, as $\NNgz$. The set of real numbers is denoted as $\RR$ and the set of extended real numbers by $\eRR = \RR \cup \{-\infty,\infty\}$, for which we use the following conventions:
\begin{enumerate}[label=(\roman*)]
    \item $x + \infty = \infty + x = \infty$ for all $x \in \RR$,
    \item $\infty + \infty = \infty$, and
    \item $x \cdot \infty = \infty \cdot x =
        \begin{cases}
            \infty  & \text{if } x \in \RRgz,\\
            0       & \text{if } x = 0,\\
            -\infty & \text{if } x \in \RRlz,
        \end{cases}$
\end{enumerate}
where $\RRgz$ denotes set of positive and $\RRlz$ the set of negative real numbers.
The conventions are defined analogously for $-\infty$. Other expressions such as $\infty-\infty$ and $\frac{\infty}{\infty}$ are undefined. 
For a given set $A$ we denote its powerset as $2^X = \{Y \mid Y \subseteq X\}$. We denote identity matrix as $\eye$.

Furthermore, we denote the entry in the $i$-th row and $j$-th column of a matrix $A$, as $(A)_{ij}$. We denote the $i$-th entry of a vector $v$ as $v(i)$.

\section{Theoretical Foundation}\label{sec:matheteil}
In this section, we introduce basic concepts from domain, measure, probability, and matrix theory, as well as other concepts used throughout this thesis.

% Domain theory
We start by introducing notions from domain theory.
The provided definitions are taken from \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens} whose material originates from \cite[Chapters 1]{Abramsky_Gabbay_Maibaum_Book_Matheteil}.

\begin{definition}[Partial Order]
    Let $X$ be a set with a relation $\sqsubseteq \subseteq X \times X$. The relation $\sqsubseteq$ is called a \textit{partial order} if the following holds for all $x,y,z \in X$:
    \begin{enumerate}[label=(\roman*)]
        \item $x \sqsubseteq x$ (reflexivity),
        \item $x \sqsubseteq y$ and $y \sqsubseteq x$ implies $x = y$ (antisymmetry), and
        \item $x \sqsubseteq y$ and $y \sqsubseteq z$ implies $x \sqsubseteq z$ (transitivity).
    \end{enumerate}
\end{definition}
A partial order $\sqsubseteq$ on a set $X$ is said to be a \textit{linear order} if any two elements in $X$ are comparable by $\sqsubseteq$.

% Measure theory
We continue by briefly presenting some concepts from measure and probability theory.
The following contents are taken from \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens} whose material originates from \cite[Chapters 1, 4, and 5]{Ash_Book_Matheteil}.
\begin{definition}[$\sigma$-Algebra, Measurable Space]
    Let $\Omega$ be a non-empty set. We call $\Sigma \subseteq 2^{\Omega}$ a \textit{$\sigma$-algebra} on $\Omega$ if the following conditions are satisfied:
    \begin{enumerate}[label=(\roman*)]
        \item $\Omega \in \Sigma$,
        \item $X \in \Sigma$ implies $\Omega\backslash X \in \Sigma$ for all $X \subseteq \Omega$, and
        \item $X_1,X_2,\dots \in \Sigma$ implies $\bigcup_{i\in \NN}X_i\in \Sigma$ for all $X_1,X_2,\dots \subseteq \Omega$.
    \end{enumerate}
    If $\Sigma$ is a $\sigma$-algebra on $\Omega$, $(\Omega,\Sigma)$ is called a \textit{measurable space}.
\end{definition}

For the extended real numbers $\eRR$, we consider the measurable space $(\eRR,\mathcal{B}(\eRR))$, where $\mathcal{B}(\eRR)$ is the \textit{Borel $\sigma$-algebra} on the extended real numbers.

\begin{definition}[Borel $\sigma$-Algebra on the Extended Real Numbers]
    The Borel $\sigma$-algebra on $\eRR$ is defined as the smallest $\sigma$-algebra $\mathcal{B}(\eRR)$ containing all intervals $(a,b] = \{x \in \eRR \mid a < x \leq b\}$ with $a,b \in \eRR$. 
\end{definition}

For a measurable space $(\Omega, \Sigma)$, a set $A \subseteq \Omega$ is called \textit{measurable} if $A \in \Sigma$. The measurability of a function is defined as follows.

\begin{definition}[(Borel) Measurable Function]
    Let $(\Omega,\Sigma)$ and $(\Omega',\Sigma')$ be measurable spaces. A function $f: \Omega \rightarrow \Omega'$ is called \textit{measurable} relative to $\Sigma$ and $\Sigma'$ if $\{\omega\in\Omega \mid f(\omega)\in f(E)\}\in \Omega$ for all $E \in \Sigma'$.

    A function $f: \Omega \rightarrow \eRR$ is called \textit{Borel measurable} (on $(\Omega,\Sigma)$) if it is measurable relative to $\Sigma$ and $\mathcal{B}(\eRR)$.
\end{definition}

Moreover, the class of Borel measurable functions is closed under addition.

\begin{theorem}[Addition of Borel Measurable Functions]
    Let $(\Omega, \Sigma)$ be a measurable space and let $f,g: \Omega \rightarrow \eRR$ be two Borel measurable functions. The function $f+g$, defined by pointwise addition, is measurable if it is well-defined, i.e., if $f(\omega) = \infty$ implies $g(\omega) \neq -\infty$ and $f(\omega) = -\infty$ implies $g(\omega) \neq \infty$ for all $\omega \in \Omega$.
\end{theorem}

Additionally, to discuss the probability of measurable sets, we introduce the notion of \textit{probability measures}.

\begin{definition}[Probability Measure, Probability Space]
    Let $(\Omega, \Sigma)$ be a measurable space. A function $\Pr : \Sigma \to [0,1]$ is a \textit{probability measure} on $(\Omega, \Sigma)$ if $\Pr(\Omega) = 1$ and for every countable collection $A_1, A_2, \ldots \in \Sigma$ of disjoint sets it holds that
    \[
        \Pr\left(\bigcup_{k \geq 1} A_k\right)
        = \sum_{k \geq 1} \Pr(A_k).
    \]
    If $\Pr : \Sigma \to [0,1]$ is a probability measure on $(\Omega, \Sigma)$, then $(\Omega, \Sigma, \Pr)$ is called a \textit{probability space}.
\end{definition}

In a probability space $(\Omega, \Sigma, \Pr)$, the elements of $\Omega$ are called outcomes, while the measurable subsets of $\Omega$ are referred to as events. An event $E \in \Sigma$ is said to hold \textit{almost surely} if $\Pr(E) = 1$.

In this thesis, we make use of the Lebesgue integral $\int_{\Omega}f(\omega)dPr(\omega)$ of a Borel measurable function $f:\Omega \rightarrow \eRR$ and a probability measure $Pr$ on a probability space $(\Omega,\Sigma)$. \begin{definition}[Expected Values]
    Let $(\Omega,\Sigma,Pr)$ be a probability space. If $f:\Omega \rightarrow \eRR$ is a Borel measurable function on $(\Omega,\Sigma,Pr)$, then the \textit{expected value} of $f$, denoted as $\Exp[f]$, is the Lebesgue integral of $f$ with respect to the probability measure $Pr$, i.e., $\Exp[f] = \int_{\Omega}f(\omega)dPr(\omega)$, provided the integral is well-defined.
\end{definition}

We continue by introducing notions from matrix theory.

An important notion, which we require in this thesis, is the \textit{(strictly) row-stochastic matrix}.
\begin{definition}[Row-Stochastic Matrix]\label{def:row-stoch_matrix}
    A square matrix $A \in \RR^{n\times n}_{\geq 0}$ is row-stochastic if and only if $\sum_{j=0}^n(A)_{ij} = 1$ for all rows $i \in \underline{n}$.
\end{definition}
Notice that if we consider each row $i$ of a row-stochastic matrix $A$ as a function $f_i(j)=(A)_{ij}$, then these functions are discrete probability distributions.

\begin{definition}[(Strictly) Row-Substochastic Matrix]\label{def:row-substoch_matrix}
    A square matrix $A \in \RR^{n\times n}_{\geq 0}$ is row-substochastic if and only if $\sum_{j=0}^n(A)_{ij} \leq 1$ for all rows $i \in \underline{n}$.\\
    $A$ is strictly row-substochastic if and only if there exists at least one row $i'$ with $\sum_{j=0}^n(A)_{i'j} < 1$
\end{definition}

If a matrix $A$ converges to the null matrix, then the inverse of $(\eye - A)$, where $\eye$ is the identity matrix, is equal to the so-called Neumann series \cite{Neumann_Matheteil}.
\begin{definition}[Matrix inversion by the Neumann Series]\label{def:neumann_series}
    Given a matrix $A\in \RR^{n\times n}$ with $\lim_{k\rightarrow \infty}A^k = 0$, i.e., $A$ is convergent, the Neumann series $\sum_{k=0}^\infty A^k$ converges against $(\eye - A)^{-1}$.
\end{definition}

We continue by introducing linear and nonlinear programming based on \cite{Luenberger_Book_Matheteil}.
\begin{definition}[Linear Programming]
    A \textit{linear programming} problem with $n$ variables and $m$ constraints can be formulated as:
    \begin{align}
    \text{minimize} \qquad & c^\T x \notag \\
    \text{s.t.} \qquad & Ax \leq b \notag \\
    & x \geq 0\ ,\notag
    \end{align}
    
    where $c,x \in \RR^n$, $A\in \RR^{m\times n}$, and $b\in \RR^m$.
\end{definition}

In contrast to linear programming, nonlinear programming has a nonlinear objective function or nonlinear constraints.
\begin{definition}[Nonlinear Programming]
    A \textit{nonlinear programming} problem with $n$ variables and $m+r$ constraints can be formulated as:
    \begin{align}
        \text{minimize} \qquad & f(x) \notag \\
        \text{s.t.} \qquad & g_i(x) \leq 0 \qquad & i \in \underline{m} \notag \\
        & h_j(x) = 0 \qquad & j \in \underline{r} \notag \\
        & x \in S\ ,\notag
    \end{align}
    
    where $x \in \RR^n$ and $f,g_i,h_j: S^n \rightarrow \RR$ for all $i\in\underline{m}$, $j\in\underline{r}$. At least one of the functions $f$, $g_i$, or $h_j$ is nonlinear.
\end{definition}

Lastly, we use the convergence of the geometric series in this thesis.
\begin{definition}[Convergence of Geometric Series]
    The geometric series $\sum_{k=0}^\infty q^k$ converges for all $q\in\RR$ with $|q|<1$, where
    \[\sum_{k=0}^\infty q^k=\frac{1}{1-q}\ ,\ \text{ for } |q|<1\]
\end{definition}
