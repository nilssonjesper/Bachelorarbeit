\begin{definition}[Gershgorin circle theorem]\label{def:gershgorin_theorem}
    Let $\tT\in \CC^{n\times n}$ be a square complex matrix, then for every eigenvalue $\lambda$ of $\tT$ it is true that 
    \[|\lambda-(\tT)_{ii}|\leq\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|\ ,\] 
    for any $i\in \underline{n}$ and where $|\cdot|$ denotes the absolute value, i.e. magnitude.\\
    In other words, all eigenvalues of $\tT$ lie on a disk in the complex plane centered at $(\tT)_{ii}$ with radius $\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|$.
\end{definition}

\begin{definition}[Row-Stochastic Matrix]\label{def:row-stoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-stochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} = 1$ for all rows $i \in \underline{n}$.
\end{definition}
When considering each row as a function $f_i(j)=(\tT)_{ij}$, these functions are discrete probability distributions.

\begin{definition}[(Strictly) Row-Substochastic Matrix]\label{def:row-substoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-substochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} \leq 1$ for all rows $i \in \underline{n}$.\\
    $\tT$ is strictly row-substochastic if and only if ther exists at least one row $i'$ with $\sum_{j=0}^n(\tT)_{i'j} < 1$
\end{definition}

\begin{theorem}\label{th:row-substoch_eigenvalue_0}
    If a matrix $\tT$ is strictly row-substochastic, then it has no eigenvalue $|\lambda| \geq 1$.
\end{theorem}

    \[
    |\lambda-(\tT)_{ii}| = |\lambda|-(\tT)_{ii} \leq \sum_{j=1\land j\neq i}^n|(\tT)_{ij}|\]
    \[ \iff 1 < |\lambda| \leq (\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij}\ .
    \]

\section{Markov Chains}
In this section, we will introduce definitions and theorems for Markov Chains which we will later require for section on Interval Markov Chains. The notation is mostly based on \cite{Bibel}, \cite{Semantics_Model-Checking_omega-Regular}, \cite{Semantics_Quant-Analysis} and \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens}.

\begin{definition}[DTMC]\label{def:DTMCs}
    A discrete-time Markov chain (DTMC) is a tuple $\MC = (S, \PF,\iota_{init})$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S$ is a countable, nonempty set of states,
        \item $\PF: S \times S \rightarrow [0,1]$ is the transition probability function, with $\sum_{s'\in S}\PF(s,s') = 1$ for all states $s \in S$, and
        \item $\iota_{init}: S \rightarrow [0,1]$ is the initial distribution, such that $\sum_{s\in S}\iota_{init}(s)=1$.
    \end{enumerate}
\end{definition}
We assume in this thesis, that the DTMCs are finite, i.e., that the set of states $S$ is finite. Furthermore, we assume an arbitrary but fixed linear order $\prec$ on the set $S$, such that the states $s\in S$ can be indexed. Thus, $s_i \prec s_j$ if $s,s' \in S$ and $i < j$.
Due to the conditions (i) and (ii) provided in \cref{def:DTMCs}, the transition probability function $\PF$ and the initial distribution $\iota_{init}$ are discrete probability distributions.

Additionally, we define a neighbour set $N$, with $s' \in N(s)$, if and only if $s'$ is a direct successor of $s$. \begin{definition}[Transient and Recurrent states]\label{def:transient_and_recurrent_states}
    A state $s \in S$ is called recurrent if and only if $Pr_s(\cCircle \Diamond s)=1$. All states that are not recurrent are called transient.
\end{definition}
In other words, a state is recurrent precisely when we are guaranteed to reach it from all of its direct successors, meaning once we reach a recurrent state, we are bound to return to it. With this characterization of EVTs for transient states, we can prove the following lemma.
\begin{lemma}[Reachability of Transient States]\label{lem:reachability_of_transient_states}
    Transient states are reachable only from other transient states. Moreover, it follows from this that if a state is reachable from a recurrent state, it must also be recurrent
\end{lemma}
We provide a short proof by contradiction
\begin{proof}
    Assume that a transient state $s_{tr}$ is reachable from a recurrent state $s_{re}$, i.e. $Pr_{s_{re}}(\cCircle\Diamond s_{tr}) > 0$. As $Pr_{s_{re}}(\cCircle\Diamond s_{re}) = 1$, see \cref{def:transient_and_recurrent_states}, necessarily $s_{re}$ is reachable from $s_{tr}$ as well and crucially $Pr_{s_{tr}}(\cCircle\Diamond s_{re}) = 1$. We now view each visit of $s_{re}$ as an indepenent trial, due to the strong Markov property \cref{def:strong_markov_property}. We can then describe the probability of not reaching $s_{tr}$ as $\lim_{N\rightarrow\infty} (1-Pr_{s_{re}}(\cCircle\Diamond s_{tr}))^N = 0$, as $s_{re}$ is visited infinitely often, see \cref{lem:EVTs_of_recurrent_states}. Thus, we are guarenteed to visit $s_{tr}$, when starting in $s_{re}$. As $Pr_{s_{tr}}(\cCircle\Diamond s_{re}) = 1$, this process repeats each time $s_{tr}$ is visited. Therefore, $s_{tr}$ is visited infinitely often as well, i.e. $\Exp[vt_{s_{tr}}] = \infty$. This contradicts the characterization of the EVTs of transient states, proven by \cite{KemenySnellBook}.
    Thus, we conclude that transient states cannot be reachable from recurrent states, i.e. can only be reached from other transient states.
\end{proof}
