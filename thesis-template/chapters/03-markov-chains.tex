\begin{definition}[DTMC]\label{def:DTMCs}
    A discrete-time Markov chain (DTMC) is a tuple $\MC = (S, \PF,\iota_{init})$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S$ is a countable, nonempty set of states,
        \item $\PF: S \times S \rightarrow [0,1]$ is the transition probability function, with $\sum_{s'\in S}\PF(s,s') = 1$ for all states $s \in S$, and
        \item $\iota_{init}: S \rightarrow [0,1]$ is the initial distribution, such that $\sum_{s\in S}\iota_{init}(s)=1$.
    \end{enumerate}
\end{definition}
We assume in this thesis, that the DTMCs are finite, i.e., that the set of states $S$ is finite. Furthermore, we assume an arbitrary but fixed linear order $\prec$ on the set $S$, such that the states $s\in S$ can be indexed. Thus, $s_i \prec s_j$ if $s,s' \in S$ and $i < j$.
Due to the conditions (i) and (ii) provided in \cref{def:DTMCs}, the transition probability function $\PF$ and the initial distribution $\iota_{init}$ are discrete probability distributions.

\begin{theorem}[k-step Transition Probability]\label{th:k-step_transition_probability}

\end{theorem}

\begin{definition}[Path of a DTMC]
    An \textit{infinite path} $\pi$ in a DTMC $\MC$ is a sequence $\pi = (s_0, s_1, \dots$) of states with $s_0,s_1,\dots \in S$, such that $\iota_{init}(\pi[0]) > 0$ \todo{(vllt $s_0$ statt $\pi[0]$ schreiben)} and $\PF(\pi[k],\pi[k+1]) > 0$, where $\pi[k]\in S$ denotes the $k$-th element of the sequence/path for all $k\in \NN_0$, i.e. $\pi[k] = s_k$. A \textit{finite path} is a finite prefix of an \textit{infinite path}.
\end{definition}

Furthermore, we define the length $|\pi|$ of a finite path $\pi = (s_0,\dots,s_k) \in Paths_{fin}^\MC$ to be equal to $k$ and for an infinite path $\pi \in \Path$ to be $|\pi| = \infty$.
The sets of all infinite and finite sets are denoted as $\Path$ and $Paths_{fin}^\MC$, respectively.

\begin{definition}[Next Step Probability]\label{def:next_step_probability}
    Let $\MC$ be a DTMC and let $E\in {\sum_{\Path}}$ be a (countable) set of infinite paths in $\MC$.
    We now define the "next step" operator, as
    \[\cCircle E = \{\pi\in\Path \mid (\pi[1],\pi[2],\dots)\in E\} \ ,\]
    i.e., a path satisfies $\cCircle E$ if and only if after removing the first state $\pi[0]$, the remaining path belongs to $E$.
    Moreover, the next step probability is defined as $Pr(\cCircle E) = \sum_{s' \in N(s)}\PF(s,s')\cdot Pr_{s'}(E)$.
\end{definition}

In this thesis, we will use the "next step" operator $\cCircle$ in combination with the "eventually" operator $\lozenge$. Specifically, we use them to denote the return probability $Pr_s(\cCircle\lozenge s)$, i.e., the probability to reach $s$ from all of its successors.

\begin{definition}[Transient and Recurrent states]\label{def:transient_and_recurrent_states}
    A state $s \in S$ is called recurrent if and only if $Pr_s(\cCircle \lozenge s)=1$. All states that are not recurrent are called transient.
\end{definition}
In other words, a state is recurrent precisely when we are guaranteed to reach it from all of its direct successors, meaning once we reach a recurrent state, we are bound to return to it. (Thus reaching it infinitely often) Using these definitions, we can split the states $S$ of a DTMC into two sets with $S_{tr}$ containing all of the transient states and $S_{re}$ containing all of the recurrent states. Furthermore, we define $\Strin \subseteq S_{in}$ with $\Strin = S_{in}\cap S_{tr}$. Additionally, we define an absorbing state.

\begin{definition}[Absorbing state]
    A states $s$ of a DTMC $\MC$ is refered to as an absorbing state if and only if $\PF^{\MC}(s,s) = 1$.
\end{definition}
This means, once the model enters an absorbing state, it cannot leave it. Notice, that absorbing states are a special type of recurrent states, as the only direct successor of an absorbing state $s\in S$ is $s$ itself and therefore trivially $Pr_s(\lozenge s)=1$.

\begin{definition}[Markov Descision Process (MDP)]\label{def:MDP}\cite{Bibel}\todo{größtenteils übernommen}
    A \textit{Markov Decision Process} (MDP) is a tuple $\MDP = (S,Act,A,\PF,\iota_{init})$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S$ is a countable set of states
        \item Act is a set of actions
        \item $A(s)\subseteq Act$ denotes the applicable actions for all states $s\in S$ 
        \item $\PF: S\times Act\times S \rightarrow [0,1]$ is the transition probability function such that $\sum_{s'\in S}\PF(s,\alpha, s') = 1$ for all $s\in S$ and $\alpha\in A(s)$

\begin{definition}[Canonical Form]\label{def:canonical_form}

        \item $S = (\PF(s,t))_{s,t\in S_{re}}$ contains the transition probabilities between the recurrent states
    \end{enumerate}
\end{definition}

Therefore, the Neumann series of $Q$ converges with $\sum^\infty_{k=0}Q^k = (\eye - Q)^{-1}$, i.e. the fundamental matrix (see \cref{def:fundamental_matrix}) is well-defined.

Furthermore, in \cite{KemenySnellBook} Kemeny and Snell prove that each entry $(F)_{ij} = \Exp_{i}[vt_j]$.
From this equation, it intuitively follows that transient states are only visited a finite amount of times.     
With this characterization of EVTs for transient states, we can prove the following lemma.
\begin{lemma}[Reachability of Transient States]\label{lem:reachability_of_transient_states}
    Transient states are reachable only from other transient states. Moreover, it follows from this that if a state is reachable from a recurrent state, it must also be recurrent
\end{lemma}
We provide a short proof by contradiction
\begin{proof}
    Assume that a transient state $s_{tr}$ is reachable from a recurrent state $s_{re}$, i.e. $Pr_{s_{re}}(\cCircle\lozenge s_{tr}) > 0$. As $Pr_{s_{re}}(\cCircle\lozenge s_{re}) = 1$, see \cref{def:transient_and_recurrent_states}, necessarily $s_{re}$ is reachable from $s_{tr}$ as well and crucially $Pr_{s_{tr}}(\cCircle\lozenge s_{re}) = 1$. We now view each visit of $s_{re}$ as an indepenent trial, due to the strong Markov property \cref{def:strong_markov_property}. We can then describe the probability of not reaching $s_{tr}$ as $\lim_{N\rightarrow\infty} (1-Pr_{s_{re}}(\cCircle\lozenge s_{tr}))^N = 0$, as $s_{re}$ is visited infinitely often, see \cref{lem:EVTs_of_recurrent_states}. Thus, we are guarenteed to visit $s_{tr}$, when starting in $s_{re}$. As $Pr_{s_{tr}}(\cCircle\lozenge s_{re}) = 1$, this process repeats each time $s_{tr}$ is visited. Therefore, $s_{tr}$ is visited infinitely often as well, i.e. $\Exp[vt_{s_{tr}}] = \infty$. This contradicts the characterization of the EVTs of transient states, proven by \cite{KemenySnellBook}.
    Thus, we conclude that transient states cannot be reachable from recurrent states, i.e. can only be reached from other transient states.
\end{proof}
