\chapter{\textit{EVTs in IDTMCs}}
\label{ch:EVTs_in_IDTMCs}

We continue in this chapter by defining \textit{expected visiting times} (EVTs) for IDTMCs. As we discussed in the previous section, IDTMCs can be seen as a set of DTMCs that satisfy the IDTMC under the given semantic. These different DTMCs can have varying EVTs for a given state $s$ in the IDTMC, thus we are interested in the upper and lower bound for the EVTs of $s$, i.e., in the largest and smallest EVT of $s$ among the satisfying DTMCs. We refer to these largest and smallest EVTs as uEVTs and $\ell$EVTs.

First, we introduce uEVTs and $\ell$EVTs of IDTMCs formally. Then, we provide some characterizations for the uEVTs and $\ell$EVTs in \Cref{sec:calculating_expected_visiting_times_of_IDTMCs}. Hereafter, we show how the uEVTs and $\ell$EVTs can be computed using quantitative reachabilities in \Cref{sec:IMC_Quantitative_Reachability} and using expected total rewards in \Cref{sec:computing-evts-using-expected-total-rewards}. Lastly, we show that the decision problem of uEVTs and $\ell$EVTs is P-complete in \Cref{sec:P-completeness}.

\begin{definition}[EVTs of IDTMCs]\label{def:EVTs_of_IDTMCs}
    For an IDTMC $\IMC$, we define the \textit{lower expected visiting time} ($\ell$EVT) $\ExpLower^\IMC[vt_s]$ and \textit{upper expected visiting time} (uEVT) $\ExpUpper^\IMC[vt_s]$ for all states $s\in S^\IMC$, as follows under UMC semantic:
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s] = \inf_{\MC\models_U\IMC} \Exp^\MC[vt_s]$,
        \item $\ExpUpper^\IMC[vt_s] = \sup_{\MC\models_U\IMC} \Exp^\MC[vt_s]$,
    \end{enumerate}
    and as follows under the IMDP semantic:
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s]
        = \inf_{\MC \models_I \IMC} \Exp^\MC[vt_T]$,
        \item $\ExpUpper^\IMC[vt_s]
        = \sup_{\MC \models_I \IMC} \Exp^\MC[vt_T]$,
    \end{enumerate}
    where $T = \tau^{-1}(s)$.
\end{definition}

\section{Quantitative Reachabilities in IDTMCs}\label{sec:relating-reach-and-EVTs}
In this section we continue by proving that given an IDTMC $\IMC$ and a state $s\in S^\IMC$ there exists at least one DTMC $\MC \models_U \IMC$, such that $\MC$ has the maximum reachability probability for all states $t\in S^\IMC$ to $s$, i.e., $Pr^{\MC}_t(s) = \Prupper^\IMC_t(s)$. Conversely, we show the same for the minimum reachability probability. 

\emph{\textbf{Remark.} Existence of DTMC with optimal reachability.}\label{rem:existence}
For any finite IDTMC $\IMC$ and states $t,s\in S^\IMC$, there exists an optimal stationary nature $\eta_{st}$ on $\IMC$ which induces a DTMC $\MC_{\eta_{st}}$, such that $Pr^{\MC_{\eta_{st}}}_t(\lozenge s) = \Prupper^{\IMC}_t(\lozenge s)$, see \cite[Lemma 10.102]{Bibel} and \cite{Badings2025}. As $\MC_{\eta_{st}} \models \IMC$, see \Cref{th:relation_of_semantics}, there thus exists a DTMC
$\MC\models\IMC$ with the maximum reachability probability from $t$ to $s$ which is precisely $\MC = \MC_{\eta_{st}}$. The same holds for the minimum reachability probability, see \cite[Lemma 10.113]{Bibel}.

Recall, the definition of intermediate states, provided in \Cref{def:intermediate_state}.
\begin{theorem}\label{th:max_reach_prob_for_intermediate_states}
    Given an IDTMC $\IMC$, states $t,s\in S^\IMC$, and a DTMC $\MC$ with $\MC\models_U \IMC$.

    If $\MC$ has the maximum reachability probability from $t$ to $s$, i.e., $Pr^{\MC}_{t}(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$, then for all $u\in I^\MC_{t,s}$:
    \[Pr^{\MC}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)\]

    Conversely, if $\MC$ has the minimum reachability probability from $t$ to $s$, i.e., $Pr^{\MC}_{t}(\lozenge s) = \Prlower^{\IMC}_{t}(\lozenge s)$, then for all $u\in I^\MC_{t,s}$:
    \[Pr^{\MC}_{u}(\lozenge s) = \Prlower^{\IMC}_{u}(\lozenge s)\]
\end{theorem}

In other words, if the reachability probability of a DTMC, under the UMC semantic, from $t$ to $s$ is optimal, then the reachability probability of all intermediate states to $s$ is optimal as well.

\begin{proof}
    % We will start by proving this theorem for $\MC \models_I \IMC$ and then show why $\MC \models_U\IMC$ follows from it.
    As stated in the previous remark, there exists at least one DTMC $\MC_{\eta}\models_U\IMC$ with $Pr_t^\MC(\lozenge s) = \Prupper^{\IMC}_t(\lozenge s)$, such that $\MC_{\eta}$ is induced by a stationary nature $\eta$ on $\IMC$. 

    Based on \cite[Theorem 10.100]{Bibel}, we can formulate the following equation which characterizes the maximum reachability probability of IDTMCs under the IMDP semantic.
    \begin{equation}\label{eq:maximum_reach_prob_MDPs}
        \Prupper^\IMC_t(\lozenge s) = \max_{\PF \in Steps^\IMC(t)} \left\{ \sum_{u\in S^\IMC} \PF(t,u) \cdot \Prupper^\IMC_u(\lozenge s)\right\} \ .
    \end{equation}

    Intuitively, the DTMCs $\MC_{\eta'}$ that are induced by natures $\eta'$ which solve this equation have the maximum reachability probability $Pr^{\MC_{\eta'}}_u(\lozenge s) = \Prupper^\IMC_u(\lozenge s)$ for all $u\in I^\MC_{t,s}$, as shown in \cite{Bibel}. As $\MC_{\eta}$ has the maximum reachability probability of $s$, when starting in $t$, and $\MC_{\eta}\models_I$, see \cref{th:relation_of_semantics}, $\MC_{\eta}$ solves \Cref{eq:maximum_reach_prob_MDPs}.

    % As discussed, a memoryless scheduler on the MDP construction induces a DTMC that satisfies the IDTMC. Because this memoryless scheduler is optimal, clearly it satisfies \Cref{eq:maximum_reach_prob_MDPs}. Thus, this theorem is proven for the UMC semantic.

    Conversely, the same can be reasoned for minimum reachability using \cite[Theorem 10.109]{Bibel}.
\end{proof}

\begin{corollary}\label{th:two_DTMCs_compatible_for_max_reach}
    Given an IDTMC $\IMC$, two transient states $t_1, t_2\in S^\IMC$, a state $s\in S^\IMC$, and two DTMCs $\MC_1,\MC_2$ with $\MC_1,\MC_2\models_U \IMC$ and $Pr^{\MC_1}_{t_1}(\lozenge s) = \Prupper^\IMC_{t_1}(\lozenge s)$ and $Pr^{\MC_2}_{t_2}(\lozenge s) = \Prupper^{\IMC}_{t_2}(\lozenge s)$. In other words the DTMCs $\MC_1$ and $\MC_2$ have the maximum reachability probability for $s$ from $t_1$ and $t_2$, respectively.

    For all $u\in I^{\MC_1}_{t_1,s} \cap I^{\MC_2}_{t_2,s}$, the reachability probabilities $Pr^{\MC_1}_u(\lozenge s) = Pr^{\MC_2}_u(\lozenge s)$ are identical.

    Conversely, the same holds if $\MC_1$ and $\MC_2$ have the minimum reachability probability for $s$ from $t_1$ and $t_2$, respectively.

\end{corollary}
In other words, if $\MC_1$ and $\MC_2$ have optimal reachability from $t_1$ and $t_2$, respectively, to $s$, then all states $u\in S^\IMC$ that are intermediate states of $t_1$ and $s$, as well as $t_2$ and $s$, have the identical reachability probability in $\MC_1$ and $\MC_2$.

\begin{proof}
    This corollary follows from \Cref{th:max_reach_prob_for_intermediate_states}.
    Using \Cref{th:max_reach_prob_for_intermediate_states}, we can conclude that $Pr^{\MC_1}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)$ for all $u \in I^{\MC_1}_{t_1,s}$ and $Pr^{\MC_2}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)$ for all $u \in I^{\MC_2}_{t_2,s}$. Thus, $Pr^{\MC_1}_u(\lozenge s) = Pr^{\MC_2}_u(\lozenge s)$ for all $u\in I^{\MC_1}_{t_1,s} \cap I^{\MC_2}_{t_2,s}$.

    Conversely, this holds for minimum reachability probabilities using the same reasoning.
\end{proof}

\begin{lemma}[One-state Extension]\label{lem:one-state_extension}
    Given an IDTMC $\IMC$, a set $T \subset S^\IMC$, a state $t \notin T$, and a state $s\in S^\IMC$, assume that $\MC_T \models_U \IMC$ with $Pr_u^{\MC_T}(\lozenge s) = \Prupper_u^\IMC(\lozenge s)$ for all $u\in T$ and $\MC_t \models_U \IMC$ with $Pr_t^{\MC_t}(\lozenge s) = \Prupper_t^\IMC(\lozenge s)$. Define $\MC_{cmb} = (S^\IMC,\PF^{\MC_{cmb}},\iota_{init}^{\MC_T})$ by
    \[\PF^{\MC_{cmb}}(w,v) = 
    \begin{cases*}
        \PF^{\MC_t}(w,v) & if $w\in I^{\MC_t}_{t,s}\backslash I^{\MC_T}_{T,s}$ \\
        \PF^{\MC_T}(w,v) & else.
    \end{cases*}\]
    Stated in words, $\MC_{cmb}$ is constructed such that its probability distribution function is equal to that of $\MC_t$ in all states that are only intermediate states for $t,s$ and not $T,s$, while it is equal to that of $\MC_T$ in all other states.
    
    Then, $Pr^{\MC_{cmb}}_u(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)$ for all $u \in T \cup \{t\}$, i.e., $\MC_{cmb}$ has the maximum reachability probability for all states in $T\cup \{t\}$.

    Conversely, if $Pr_u^{\MC_T}(\lozenge s) = \Prlower_u^\IMC(\lozenge s)$ for all $u\in T$ and $Pr_t^{\MC_t}(\lozenge s) = \Prlower_t^\IMC(\lozenge s)$, the identical construction of $\MC_{cmb}$ implies $Pr^{\MC_{cmb}}_u(\lozenge s) = \Prlower^{\IMC}_{u}(\lozenge s)$ for all $u \in T \cup \{t\}$.
\end{lemma}

Intuitively, this lemma states that using two DTMCs $\MC_T$ and $\MC_t$ which have optimal reachability probabilities from $T$ and $t$ to $s$, respectively, we can construct a DTMC $\MC_{cmb}$ that has optimal reachability from $T\cup\{t\}$ to $s$.

\begin{proof}
    First, as for each $w\in S^\IMC$ either $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{t}}(w,v)$ or $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{T}}(w,v)$ for all $v\in S^\IMC$, it follows that $\PF^{\MC_{cmb}}$ is well-defined, i.e., $\PF^{\MC_{cmb}}$ is a discrete probability distribution, see \Cref{def:discrete_probability_distribution}. Next, as $\MC_t,\MC_T \models_U \IMC$ it follows from the construction of $\MC_{cmb}$ that also $\MC_{cmb} \models_U \IMC$.
    
    %Recall, that $Pr^{\MC_{cmb}}_{}(\lozenge s) = \sum_{u\in N(t)\cap I^{\MC_{cmb}}_{t,s}}\PF^{\MC_{cmb}}(t,u)\cdot Pr_u^{\MC_{cmb}}(\lozenge s)$, see \Cref{lem:recursive_formula_for_reachability_prob}. 
    As $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{T}}(w,v)$ for all $w \in I^{\MC_T}_{T,s}$, we can conclude that $Pr^{\MC_{cmb}}_u(\lozenge s) = Pr^{\MC_{T}}_u(\lozenge s)$ for all $u\in T$, as \Cref{lem:recursive_formula_for_reachability_prob} implies that the reachability probability from $u$ to $s$ only depends on the intermediate states of $T$ and $s$. Thus, $\Prupper^{\IMC}_{u}(\lozenge s) = Pr^{\MC_{cmb}}_u(\lozenge s)$ for all $u\in T$, by definition of $\MC_{T}$.

    Thus, it remains to show that $Pr^{\MC_{cmb}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$. As we know that $Pr^{\MC_{t}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$ by definition, it suffices to prove that $Pr^{\MC_{cmb}}_t(\lozenge s) = Pr^{\MC_{t}}_t(\lozenge s)$.
    % Notice, that if $I^{\MC_t}_{t,s} \cap I^{\MC_T}_{T,s} = \emptyset$, then the proof concludes using the same argument as for $u\in T$. Furthermore, if $I^{\MC_t}_{t,s}\subseteq I^{\MC_T}_{T,s}$, then the proof concludes directly using \Cref{th:max_reach_prob_for_intermediate_states}.
    % \todo{sollen diese extra anmerkungen drin bleiben?}

    Using \Cref{lem:recursive_formula_for_reachability_prob}, it follows that 
    \[Pr^{\MC_{cmb}}_{t}(\lozenge s) = \sum_{u\in N(t)\cap I^{\MC_{cmb}}_{t,s}}\PF^{\MC_{cmb}}(t,u) Pr_u^{\MC_{cmb}}(\lozenge s)\ .\]
    For all $u\in I^{\MC_t}_{t,s}\backslash I^{\MC_T}_{T,s}$, $\PF^{\MC_{cmb}}(u,w) = \PF^{\MC_{t}}(u,w)$ and for all $u\in I^{\MC_T}_{T,s} \cap I^{\MC_t}_{t,s}$, it holds that $Pr^{\MC_{T}}_u(\lozenge s) = Pr^{\MC_t}_u(\lozenge s)$, see \Cref{th:two_DTMCs_compatible_for_max_reach}, which implies $Pr^{\MC_{cmb}}_u(\lozenge s) = Pr^{\MC_t}_u(\lozenge s)$. From this, we can conclude that $Pr^{\MC_{t}}_{t}(\lozenge s) = Pr^{\MC_{cmb}}_{t}(\lozenge s)$.

    Thus, we have shown that $Pr_u^{\MC_{cmb}}(\lozenge s) = \Prupper_u^{\IMC}(\lozenge s)$ for all $u\in T \cup \{t\}$.

    Conversely, the same holds for minimum reachability probabilities, given that $Pr_u^{\MC_T}(\lozenge s) = \Prlower_u^\IMC(\lozenge s)$ for all $u\in T$ and $Pr_t^{\MC_t}(\lozenge s) = \Prlower_t^\IMC(\lozenge s)$.
\end{proof}

Crucially, \Cref{lem:one-state_extension} can be extended to all states $s\in S^\IMC$.
\begin{theorem}\label{th:existence_of_dtmc_with_universal_max_reach}
    Given an IDTMC $\IMC$ and a state $s\in S^\IMC$ there exists a DTMC $\MC_{max} \models_U \IMC$, such that $Pr_t^{\MC_{max}}(\lozenge s) = \Prupper^\IMC_t(\lozenge s)$ for all states $t\in S^\IMC$.

    Conversely, there exists a DTMC $\MC_{min} \models_U \IMC$, such that $Pr_t^{\MC_{min}}(\lozenge s) = \Prupper^\IMC_t(\lozenge s)$ for all states $t\in S^\IMC$.
\end{theorem}

\begin{proof}
    Consider any given enumeration $t_1, t_2, \dots$ of $S^\IMC$. Clearly, this enumeration is finite, since $S^\IMC$ is finite, see \Cref{assmp:finiteness_IDTMCs}.
    
    \emph{Construction.} Now, let $\MC_{max}^{(1)}\models_U \IMC$ with $Pr_{t_1}^{\MC_{max}^{(1)}}(\lozenge s) = \Prupper^\IMC_{t_1}(\lozenge s)$ and $T_1 \coloneq I^{\MC_{max}^{(1)}}_{t_1,s}$. By \Cref{th:max_reach_prob_for_intermediate_states}, $Pr_u(\lozenge s) = \Prupper_u(\lozenge s)$ for all $u\in I^{\MC_{max}^{(1)}}_{t_1,s}$.

    Inductively, $\MC_{max}^{(n)} \models_U \IMC$ and $s\in T_n = I^{\MC_{max}^{(n)}}_{t_n,s}$, see \Cref{def:intermediate_state}.
    If $t_{n+1}\in T_n$, then we define $\MC_{max}^{(n+1)} \coloneq \MC_{max}^{(n)}$ and $T_{n+1} \coloneq T_n$. If not, we use \Cref{lem:one-state_extension} with 
    \begin{itemize}
        \item $T = T_n$,
        \item $\MC_T = \MC_{max}^{(n)}$,
        \item $t = t_{n+1}$, and
        \item $\MC_t = \MC$ for some $\MC \models_U \IMC$ with $Pr^\MC_{t+1}(\lozenge s) = \Prupper_{t+1}^\IMC(\lozenge s)$
    \end{itemize}
    to obtain $\MC_{max}^{(n+1)} \coloneq \MC_{cmb}$ with $\MC_{max}^{(n+1)}\models_U \IMC$. We set $T_{n+1} = T_n \cup I^{\MC_{max}^{(n+1)}}_{t_{n+1},s}$ and thus, $Pr_{u}^{\MC_{max}^{(n+1)}}(\lozenge s) = \Prupper_{u}^\IMC(\lozenge s)$ for all $u \in T_n\cup\{t_{n+1}\}$, by \Cref{lem:one-state_extension}, as $\MC_{max}^{(n+1)} = \MC_{cmb}$, and $Pr_{u}^{\MC_{max}^{(n+1)}}(\lozenge s) = \Prupper_{u}^\IMC(\lozenge s)$ for all $u\in I^{\MC_{max}^{(n+1)}}_{t_{n+1},s}$ see \Cref{th:max_reach_prob_for_intermediate_states}. Therefore, $Pr_{t_i}^{\MC_{max}^{(n+1)}}(\lozenge s) = \Prupper_{t_i}^\IMC(\lozenge s)$ for all $t_i \in T_{n+1}$. Hence, $\MC_{max} \coloneq \MC_{max}^{(|S^\IMC|)}$ has maximum reachability to $s$ for all states $t\in S^\IMC$.

    Conversely, the same holds for $\MC_{min}$, by choosing the DTMC $\MC_{min}^{(1)}$, such that $Pr^{\MC_{min}^{(1)}}_{t_1}(\lozenge s) = \Prlower_{t_1}^\IMC(\lozenge s)$ and $\MC_t = \MC$ for some $\MC \models_U \IMC$ with $Pr^\MC_t(\lozenge s) = \Prlower_t^\IMC(\lozenge s)$, when using \Cref{lem:one-state_extension} in the construction.
\end{proof}
Thus, to conclude, for any given IDTMC $\IMC$ and state $s\in S^\IMC$, there exist the DTMCs $\MC_{max},\MC_{min} \models_U \IMC$ with the maximum and minimum reachability probability to $s$, respectively, for all states $t\in S^\IMC$.

Notice that the most intuitive upper and lower bounds for the uEVTs and $\ell$EVTs of IDTMCs under the UMC semantic can be derived by using \Cref{eq:EVT_DTMC_iterative_characterization} and the upper and lower transition probability functions, as well as the upper and lower initial distributions defined in \Cref{def:IMC}.

\begin{theorem}[Upper and Lower Bounds for uEVTs and $\ell$EVTs]\label{th:loose_bounds_for_evts}
    For a given IDTMC $\IMC$ and state $s\in S^\IMC$, the uEVT of $s$ is bounded from above by 
    \[\ExpUpper^\IMC[vt_s] \leq \Exp^\IMC_{upp}[vt_s] \coloneq \Initupper(s)+\sum_{t\in S}\PFupper(t,s)\cdot \Exp^\IMC_{upp}[vt_t]\]
    and the $\ell$EVT of $s$ is bounded from below by 
    \[\ExpLower^\IMC[vt_s] \geq \Exp^\IMC_{low}[vt_s] = \Initlower(s)+\sum_{t\in S}\PFlower(t,s)\cdot \Exp^\IMC_{low}[vt_t]\]
    for all $s\in S^\IMC$ under the UMC semantic.
\end{theorem}

\emph{\textbf{Remark.} Non-attainability of values in EVT intervals under the UMC semantic.}

\section{Computing EVTs using Quantitative Reachabilities}\label{sec:IMC_Quantitative_Reachability}


Similar to \Cref{sec:MC_Quantitative_Reachability}, we relate \textit{expected visiting times} to \textit{quantitative reachabilities} in this section. Moreover, we show that for any IDTMC $\IMC$ and state $s\in S^\IMC$, there exist at least two DTMCs $\MC_{max},\MC_{min}\models \IMC$ that have the optimal, i.e., maximum or minimum, reachability probability to $s$ for all states in $\IMC$. We then demonstrate how the $\ell$EVTs and uEVTs can be calculated under the UMC semantic by using the quantitative reachabilities of these two DTMCs $\MC_{max}$ and $\MC_{min}$. 
Lastly, we show that the UMC and IMDP semantic coincide on the $\ell$EVTs and uEVTs for all states $s\in S^\IMC$. 

%Given the quantitative reachabilities of the DTMCs that satisfy a certain IDTMC $\IMC$, we can calculate the corresponding EVTs of a given state $s$ in $\IMC$ and thus find the lEVTs and uEVTs using the results from \Cref{sec:MC_Quantitative_Reachability}.

As we concluded in \Cref{sec:MC_Quantitative_Reachability}, we can calculate the EVT for all transient states $s \in S_{tr}^\MC$ in a DTMC $\MC$ using quantitative reachabilities, see \Cref{lem:EVT_via_quant_reach}: $\Exp^\MC[vt_s] = \frac{Pr^\MC(\lozenge s)}{1-Pr^\MC_s(\Circle\lozenge s)}$. As defined in \Cref{def:EVTs_of_IDTMCs}, the $\ell$EVT and uEVT of $\IMC$, under the UMC semantic, are equal to the largest and smallest EVT  amongst the satisfying DTMCs $\MC \models_U \IMC$. Thus, the $\ell$EVT and uEVT of $\IMC$ are given by the smallest and largest ratio of these quantitative reachability probabilites, see \Cref{lem:EVT_via_quant_reach}, among the satisfying Markov Chains, as those optimal ratios define the smallest and largest EVT.

\begin{theorem}\label{th:u/lEVTs_via_reach_and_semantics_coincide}
    Given an IDTMC $\IMC$, state $s\in S^\IMC$, and the corresponding DTMCs $\MC_{max},\MC_{min}\models_U \IMC$:
    \begin{enumerate}[label=(\roman*)]
        \item There exists no $\MC \models_U \IMC$ such that $\Exp^{\MC_{max}}[vt_s] < \Exp^{\MC}[vt_s]$.
        
        Conversely, there exists no $\MC \models_U \IMC$ such that $\Exp^{\MC_{min}}[vt_s] > \Exp^{\MC}[vt_s]$. 

        In other words $\Exp^{\MC_{max}}[vt_s] = \ExpUpper^\IMC[vt_s]$ and $\Exp^{\MC_{min}}[vt_s] = \ExpLower^\IMC[vt_s]$ under the UMC semantic.
        % The optimal ratios are given by 
        % \[\Prupper^\IMC(\lozenge s) \text{ and } \Prupper^\IMC_s(\cCircle\lozenge s) \qquad \text{or} \qquad \Prlower^\IMC(\lozenge s) \text{ and } \Prlower^\IMC_s(\cCircle\lozenge s)\ .\]
        % In other words, the $\ell$EVT and uEVT of $s$ can be calculated using these optimal reachability probabilities

        \item The $\ell$EVT and uEVT of $s$ are identical under both semantics, i.e., the choice of semantics does not affect the value of the $\ell$EVT and uEVT.
    \end{enumerate}
\end{theorem}

\begin{proof}\label{pr:u/lEVTs_via_reach_and_semantics_coincide}
    We start by proving claim (i).

    As we showed in \Cref{th:existence_of_dtmc_with_universal_max_reach}, there exist DTMCs $\MC_{max}, \MC_{min} \models_U \IMC$ for any given IDTMC $\IMC$ and state $s\in S^\IMC$ that have maximum or minimum reachability from all states to $s$, i.e., 
    \[Pr^{\MC_{max}}_t(\lozenge s) = \Prupper^\IMC_t(\lozenge s) \qquad \text{or} \qquad Pr^{\MC_{min}}_t(\lozenge s) = \Prlower^\IMC_t(\lozenge s)\] 
    for all $t\in S^\IMC$. We fix these two DTMCs for rest of this proof. Further, we can thus conclude that 
    \begin{equation}\label{eq:Pr_of_max_equals_Prupper}
        Pr^{\MC_{max}}(\lozenge s) = \Prupper^\IMC(\lozenge s) \qquad \text{and} \qquad Pr^{\MC_{max}}_s(\cCircle\lozenge s) = \Prupper^\IMC_s(\cCircle \lozenge s)
    \end{equation}
    or 
    \begin{equation}\label{eq:Pr_of_min_equals_Prlower}
        Pr^{\MC_{min}}(\lozenge s) = \Prlower^\IMC(\lozenge s) \qquad \text{and} \qquad Pr^{\MC_{min}}_s(\cCircle\lozenge s) = \Prlower^\IMC_s(\cCircle \lozenge s)\ .
    \end{equation}

    Using \Cref{lem:EVT_via_quant_reach}, we can thus differentiate the following three cases for $s$:
    \begin{enumerate}
        \item If $s \in S^\IMC_{\forall re}$, then $\Prupper^\IMC_s(\cCircle\lozenge s) = \Prlower^\IMC_s(\cCircle\lozenge s) = 1$, by definition of $S^\IMC_{\forall re}$, and thus \Cref{lem:EVT_via_quant_reach} is undefined for $\MC_{max}$ and $\MC_{min}$, as $Pr^{\MC_{max}}_s(\cCircle\lozenge s) = Pr^{\MC_{min}}_s(\cCircle\lozenge s) = 1$. 

        However, using \Cref{th:Expupper_ExpLower_infty} we can conclude that $\ExpUpper[vt_s] = \ExpLower[vt_s] = \infty$. By the definition of $S^\IMC_{\forall re}$, we can deduce that $s$ is reachable and recurrent in both $\MC_{max}$ and $\MC_{min}$. Thus, using \Cref{lem:EVTs_of_recurrent_states}, we can conclude that $\Exp^{\MC_{max}}[vt_s] = \ExpUpper[vt_s] = \infty$ and $\Exp^{\MC_{min}}[vt_s] = \ExpLower[vt_s] = \infty$.

        \item If $s\in S^\IMC_{re}$, then $\Prupper^\IMC_s(\cCircle\lozenge s) = 1$, by definiton of $S_{re}^\IMC$ see \Cref{def:transient_recurrent_states_IDTMC}, and thus \Cref{lem:EVT_via_quant_reach} is undefined for $\MC_{max}$, as $Pr^{\MC_{max}}_s(\cCircle\lozenge s) = 1$. However, using \Cref{th:Expupper_ExpLower_infty} we can conclude that $\ExpUpper[vt_s] = \infty$. Using \Cref{eq:Pr_of_max_equals_Prupper} and \Cref{assmp:finiteness_IDTMCs} which defines that $s$ is reachable in $\IMC$, we can deduce that $s$ is reachable and recurrent in $\MC_{max}$. Thus, using \Cref{lem:EVTs_of_recurrent_states}, it follows that $\Exp^{\MC_{max}}[vt_s] = \ExpUpper[vt_s] = \infty$. For the calculation of $\ExpLower[vt_s]$, see the equations in case (3) using that $s\notin S^\IMC_{\forall re}$ (see case 1) and thus there exists a DTMC $\MC \models_U$ with $Pr^\MC_s(\cCircle\lozenge s) < 1$, i.e., $Pr^{\MC_{min}}_s(\cCircle\lozenge s) = \Prlower^{\IMC}_s(\cCircle\lozenge s) < 1$. Hence, the equations in (3) are well-defined for $\ExpLower[vt_s]$ and $s\in S^\IMC_{re}$.
        
        \item In all remaining cases, we can calculate the $\ell$EVT and uEVT using \Cref{lem:EVT_via_quant_reach}:
        \[ \ExpLower[vt_s] = \frac{\Prlower^\IMC(\lozenge s)}{1-\Prlower^\IMC_s(\cCircle\lozenge s)} = \frac{Pr^{\MC_{min}}(\lozenge s)}{1-Pr^{\MC_{min}}_s(\cCircle\lozenge s)} = \Exp^{\MC_{min}}[vt_s]\]
        and
        \[ \ExpUpper[vt_s] = \frac{\Prupper^\IMC(\lozenge s)}{1-\Prupper^\IMC_s(\cCircle\lozenge s)} = \frac{Pr^{\MC_{max}}(\lozenge s)}{1-Pr^{\MC_{max}}_s(\cCircle\lozenge s)} = \Exp^{\MC_{max}}[vt_s]\ .\]
        As $s\notin S_{re}^\IMC$ (see case 1 and 2), by definition, there exists no DTMC $\MC \models_U \IMC$, such that $Pr^\MC_s(\cCircle\lozenge s) = 1$. In other words, $Pr^\MC_s(\cCircle\lozenge s) < 1$ for all $\MC \models_U \IMC$. Thus, the equations are well-defined.
    \end{enumerate}

    We continue by shortly proving the equations presented in case (3) by contradiction. Let us assume that there exists a DTMC $\MC \models_U \IMC$ such that $\Exp^{\MC_{max}}[vt_s] < \Exp^{\MC}[vt_s]$. Using \Cref{lem:EVT_via_quant_reach}, it thus follows that either 
    \[Pr^\MC(\lozenge s) > Pr^{\MC_{max}}(\lozenge s) \qquad \text{or} \qquad Pr^\MC(\cCircle\lozenge s) > Pr^{\MC_{max}}(\cCircle\lozenge s)\ .\]
    However, as $Pr^{\MC_{max}}(\lozenge s) = \Prupper^{\IMC}(\lozenge s)$ and $Pr^{\MC_{max}}(\cCircle\lozenge s) = \Prupper^ {\IMC}(\cCircle\lozenge s)$ this leads to a contradiciton.
    The proof for $\MC_{min}$ follows analogously by reversing the inequalities.

    Lastly, we prove (ii).
    Recall, that for any DTMC $\MC \models_U \IMC$, then $\MC \models_I \IMC$, see \Cref{th:relation_of_semantics}. Thus, $\MC_{max},\MC_{min} \models_I \IMC$.     \end{aligned}

where $(\operatorname{opt},\sim) \in \{(\max,\leq),(\min,\geq)\}$.

We now show that $\MC^s_{aug}$ satisfies $\IMC$ under the IMDP semantic.
\begin{lemma}
    Given and IDTMC $\IMC$, a state $s\in S^\IMC$, and the corresponding $\MC_{aug}^s$, it holds that $\MC_{aug}^s\models_{I} \IMC$.
\end{lemma}

\begin{proof}
    We show that $\MC_{aug}^s$ is induced by the nature
    $\eta_{aug}$ on $\IMC$ defined by
    \[\eta_{aug}(\pi) = \begin{cases*}
        \PF^{\MC_{s_{in}\rightarrow s}}(last(\pi)) & if $\neg\exists k \in \underline{|\pi|}$ with $\pi[k] = s$\\
        \PF^{\MC_{s\rightarrow s}}(last(\pi)) & otherwise.
    \end{cases*}\]
    Since $\MC_{s_{in}\rightarrow s},\MC_{s\rightarrow s}\models_I\IMC$, both transition
    distributions are admissible in $\IMC$. Hence
    $\eta_{aug}$ is a valid nature on $\IMC$.

    Before the first visit to $s$, $\eta_{aug}$ chooses the
    transition probabilities of $\MC_{s_{in}\rightarrow s}$. Therefore,
    for all $s_i^{(1)},s_j^{(1)}\in S_{st1}$ with $s_i\neq s$,
    \[
        \PF^{\MC_{aug}^s}(s_i^{(1)},s_j^{(1)})
        =
        \mu(s_j)
        =
        \PF^{\MC_{s_{in}\rightarrow s}}(s_i,s_j)
    \]
    with $\mu = \eta_{aug}(\pi)$ for every finite path $\pi$ in $\IMC$ ending in $s_i$ that does not contain
    $s$.
    This equation holds by the construction of $\MC_{aug}^s$ and $\eta_{aug}$. 

    If the transition reaches $s$, the augmented DTMC enters the
    second stage. After the first visit to $s$, $\eta_{aug}$ chooses the
    transition probabilities of $\MC_{s\rightarrow s}$. Consequently,
    for all $s_i^{(2)},s_j^{(2)}\in S_{st2}$,
    \[
        \PF^{\MC_{aug}^s}(s_i^{(2)},s_j^{(2)})
        =
        \mu(s_j)
        =
        \PF^{\MC_{s\rightarrow s}}(s_i,s_j)
    \]
    with $\mu = \eta_{aug}(\pi)$ for every finite path $\pi$ in $\IMC$ ending in $s_i$ that contains $s$. This equation holds by the construction of $\MC_{aug}^s$ and $\eta_{aug}$. 

    Finally, the initial distribution of $\MC_{aug}^s$ is obtained from
    the initial distribution of $\MC_{s_{in}\rightarrow s}$ by assigning
    it to the first stage copies of the states for all $t\in S_{tr}^{\MC_{s_{in}\rightarrow s}}$. Hence, the initial distribution agrees with that induced by $\eta_{\mathrm{aug}}$.

    Thus, $\MC_{aug}^s$ is precisely the DTMC induced by
    $\eta_{\mathrm{aug}}$ on $\IMC$. Therefore,
    \[
        \MC_{aug}^s\models_I\IMC.
    \]
\end{proof}

\section{Computing EVTs using Expected Total Rewards}\label{sec:computing-evts-using-expected-total-rewards}
In this section we examine how the $\ell$EVTs and uEVTs of an IDTMC $\IMC$ can be calculated using expected total rewards. For this, we start by formally introducing so-called Bounded-Parameter Markov Decision Processes (BMDPs). We then continue, by providing the construction of a BMDP, for any given IDTMC, such that the probability measures of the satisfying MDPs of the BMDP and the satisfying DTMCs of the IDTMC under the UMC semantic, coincide.
%\todo{von chatty besseren satz machen lassen}

Recall the definition of MDPs, provided in \Cref{def:MDP}. Similar to IDTMCs, so-called Bounded-Parameter Markov Decision Processes (BMDPs) were introduced to capture potential imprecisions of MDPs. We define these BMDPs based on \cite{Givan_Bounded_Parameter_MDP_discounted,Tewari_Bounded_Parameter_MDP} using the notation of \cite{Bibel}.

\begin{definition}[Bounded-Parameter Markov Decision Processes]
    A \textit{Bounded-Paramter Markov Decision Process} (BMDP) is a tuple \\$\BMDP = (S^\BMDP,Act^\BMDP,\PFlower^\BMDP,\PFupper^\BMDP,s_0^\BMDP,\rewLower^\BMDP,\rewUpper^\BMDP)$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S^\BMDP$ is a countable set of states,
        \item $Act^\BMDP$ is a set of actions,
        \item $\PFlower^\BMDP: S^\BMDP \times Act^\BMDP \times S^\BMDP \rightarrow [0,1]$ is the lower bound on the transition probability function,
        \item $\PFupper^\BMDP: S^\BMDP \times Act^\BMDP \times S^\BMDP \rightarrow [0,1]$ is the upper bound on the transition probability function,
        \item $s_0^\BMDP$ is the fixed initial state,
        \item $\rewLower^\BMDP: S^\BMDP \rightarrow\RR_{\geq 0}$ is the lower bound on the reward function,
        \item $\rewUpper^\BMDP: S^\BMDP \rightarrow\RR_{\geq 0}$ is the upper bound on the reward function,
    \end{enumerate}
    such that $\PFlower^\BMDP(s_i,\alpha,s_j) \leq \PFupper^\BMDP(s_i,\alpha,s_j)$ for all $s_i,s_j\in S^\BMDP$, $\alpha \in Act^\BMDP$, as well as $\rewLower^\BMDP(s) \leq \rewUpper^\BMDP(s)$ for all $s\in S^\BMDP$.
\end{definition}
Whenever the BMDP $\BMDP$ is evident from the context, we will omit the superscript $\BMDP$.

For the rest of this section, we fix a BMDP $\BMDP = (S^\BMDP,Act^\BMDP,\PFlower^\BMDP,\PFupper^\BMDP,s_0^\BMDP,\rewLower^\BMDP,\rewUpper^\BMDP)$ and an MDP $\MDP = (S^\MDP,Act^\MDP,\PF^\MDP,s_0^\MDP,rew^\MDP)$.
Similarly to IDTMCs, a BMDP $\BMDP$ defines a set of satisfying MDPs $\MDP$. An MDP $\MDP$ satisfies $\BMDP$ if and only if 
\begin{itemize}
    \item $S^\MDP = S^\BMDP$,
    \item $Act^\MDP = Act^\BMDP$,
    \item $\PF^\MDP(s_i,\alpha,s_j) \in [\PFlower^\BMDP(s_i,\alpha,s_j),\PFupper^\BMDP(s_i,\alpha,s_j)]$ for all $s_i,s_j\in S^\MDP$, $\alpha\in Act^\MDP$, and
    \item $rew^\MDP(s) \in [\rewLower^\BMDP(s),\rewUpper^\BMDP(s)]$ for all $s\in S^\MDP$,
\end{itemize}
see \cite{Tewari_Bounded_Parameter_MDP}. Notice, that this is simply the UMC semantic for BMDPs, see \Cref{def:once-and-for-all_semantic}. We denote that an MDP $\MDP$ satisfies a BMDP $\BMDP$, as $\MDP \models \BMDP$.

Furthermore, we define the minimum and maximum expected total rewards of BMDPs as follows.

\begin{definition}[Expected Total Rewards of BMDPs]\label{def:expected-total-rewards-of-bmpds}
    For a BMDP $\BMDP$, we define the \textit{minimum expected total reward} $\ExpLower^\BMDP[tr_{rew}]$ and \textit{maximum expected total reward} $\ExpUpper^\BMDP[tr_{rew}]$ for all states $s\in S^\BMDP$, as follows:
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\BMDP[tr_{rew}] = \inf_{\MDP\models\BMDP} \Exp^\MDP[tr_{rew}]$,
        \item $\ExpUpper^\BMDP[tr_{rew}] = \sup_{\MDP\models\BMDP} \Exp^\MDP[tr_{rew}]$,
    \end{enumerate}
\end{definition}

We now fix an IDTMC $\IMC = (S^\IMC,\PFlower^\IMC,\PFupper^\IMC,s_0)$ under the fixed-initial-state definition. From this IDTMC $\IMC$, given a state $s\in S^\IMC$, we continue by constructing the following BMDP $\BMDP_s$.

\emph{\textbf{Construction.}} We define the BMDP $\BMDP_s = (S^\IMC,Act^{\BMDP_s},\PFlower^{\BMDP_s},\PFupper^{\BMDP_s},s_0^\IMC,\rewLower^{\BMDP_s},\rewUpper^{\BMDP_s})$ with $Act^{\BMDP_s} = \{\alpha\}$,
\[\PFlower^{\BMDP_s}(s_i,\alpha,s_j) = \PFlower^\IMC(s_i,s_j) \qquad \text{and} \qquad \PFupper^{\BMDP_s}(s_i,\alpha,s_j) = \PFupper^\IMC(s_i,s_j)\]
for all $s_i,s_j \in S^\IMC$.
Furthermore,
\[
\rewLower^{\BMDP_s}(s_i)
=
\rewUpper^{\BMDP_s}(s_i)
=
\llbracket s_i = s \rrbracket
\]
for all $s_i \in S^\IMC$.

\begin{lemma}[Correspondence between satisfying DTMCs and MDPs]\label{lem:correspondance-between-satisfying-dtmcs-and-mdps}
\label{lem:DTMC_MDP_correspondence}
Let $\IMC = (S^\IMC,\PFlower^\IMC,\PFupper^\IMC,s_0)$ be an IDTMC under the fixed-initial-state definition, $s\in S^\IMC$, and let $\BMDP_s$ be the corresponding BMDP as constructed above. Then, for every DTMC $\MC$, $\MC\models_U\IMC$ if and only if there exists an MDP $\MDP$ such that $\MDP\models\BMDP_s$ and the probability measures induced by $\MC$ and $\MDP$ correspond under the identification of their paths.
\end{lemma}

We provide a proof for this lemma.
\begin{proof}
First, let $\MC=(S^\MC,\PF^\MC,s_0^\MC)$ be a DTMC such that $\MC\models_U\IMC$. We construct an MDP $\MDP=(S^\MDP,Act^\MDP,\PF^\MDP,s_0^\MDP,rew^\MDP)$ by defining $S^\MDP=S^\IMC$, $Act^\MDP=\{\alpha\}$, $\PF^\MDP(s_i,\alpha,s_j)=\PF^\MC(s_i,s_j)$ for all $s_i,s_j\in S^\IMC$, and $s_0^\MDP=s_0^\MC$. Furthermore, let 
\[rew^\MDP(s_i)= \llbracket s_i = s \rrbracket\]
for $s\in S^\IMC$.

Since $\MC\models_U\IMC$, we have $\PFlower^\IMC(s_i,s_j)\leq\PF^\MC(s_i,s_j)\leq\PFupper^\IMC(s_i,s_j)$ for all $s_i,s_j\in S^\IMC$. By the construction of $\BMDP_s$, this implies
\[
\PFlower^{\BMDP_s}(s_i,\alpha,s_j)\leq\PF^\MDP(s_i,\alpha,s_j)\leq\PFupper^{\BMDP_s}(s_i,\alpha,s_j).
\]
Moreover, $\rewLower^{\BMDP_s}(s_i)=rew^\MDP(s_i)=\rewUpper^{\BMDP_s}(s_i)=\llbracket s_i = s \rrbracket$. Thus, $\MDP\models\BMDP_s$.

Conversely, let $\MDP=(S^\MDP,Act^\MDP,\PF^\MDP,s_0^\MDP,rew^\MDP)$ be an MDP such that $\MDP\models\BMDP_s$. Since $Act^{\BMDP_s}=\{\alpha\}$, we have $Act^\MDP=\{\alpha\}$. We construct a DTMC $\MC=(S^\MC,\PF^\MC,s_0^\MC)$ by defining $S^\MC=S^\MDP=S^\IMC$, $\PF^\MC(s_i,s_j)=\PF^\MDP(s_i,\alpha,s_j)$ for all $s_i,s_j \in S^\MDP$, and choosing $s_0^\MC = s_0^\MDP$.

Since $\MDP\models\BMDP_s$, it follows that
\[
\PFlower^\IMC(s_i,s_j)=\PFlower^{\BMDP_s}(s_i,\alpha,s_j)\leq\PF^\MDP(s_i,\alpha,s_j)\leq\PFupper^{\BMDP_s}(s_i,\alpha,s_j)=\PFupper^\IMC(s_i,s_j).
\]
Hence, $\PF^\MC(s_i,s_j)\in[\PFlower^\IMC(s_i,s_j),\PFupper^\IMC(s_i,s_j)]$ for all $s_i,s_j\in S^\IMC$. Together with the fixed initial state $s_0$, we can thus conclude that $\MC\models_U\IMC$.

It remains to show that the induced probability measures correspond. To this end, let $\pi=(s_0,s_1,\dots,s_n)\in Paths^\MC_{fin}$ be a finite path of $\MC$. Since $\MDP$ has exactly one action, $\pi$ corresponds to the MDP path $\hat{\pi}=(s_0,\alpha,s_1,\alpha,\dots,\alpha,s_n) \in Paths^\MDP_{fin}$. This correspondence is a bijection between the paths of $\MC$ and the paths of $\MDP$. For every finite path $\pi = (s_0,s_1,\dots,s_n) \in Paths^\MC_{fin}$, the corresponding cylinder probabilities are
\[
Pr^\MC(Cyl(\pi))=\llbracket \pi[0] = s_0^\MC\rrbracket \cdot \prod_{k=0}^{n-1}\PF^\MC(\pi[k],\pi[k+1])
\]
and
\[
Pr^\MDP(Cyl(\hat{\pi}))=\llbracket \pi[0] = s_0^\MDP\rrbracket \cdot\prod_{k=0}^{n-1}\PF^\MDP(\pi[k],\alpha,\pi[k+1]).
\]
By construction, $s_0^\MC = s_0^\MDP$ and $\PF^\MC(s_i,s_j)=\PF^\MDP(s_i,\alpha,s_j)$. Consequently, it follows that $\Pr^\MC(\pi)=\Pr^\MDP(\hat{\pi})$ for every finite path $\pi\in Paths^\MC_{fin}$ and $\hat{\pi}\in Paths^\MDP_{fin}$. Hence, the corresponding cylinder sets have the same probability, and thus the probability measures induced by $\MC$ and $\MDP$ correspond under the above bijection.
\end{proof}

Notice that the reward function $rew^\MDP$ of all MDPs $\MDP \models \BMDP_{s}$ is precisely the reward function defined in \Cref{lem:from-expected-rewards-to-evts} with $x = 1$. Moreover, we can conclude from \Cref{lem:correspondance-between-satisfying-dtmcs-and-mdps} that there exists a corresponding DTMC $\MC \models_U \IMC$ for each $\MDP$, such that $\Exp^\MC[tr_{rew}] = \Exp^\MDP[tr_{rew}]$, as their probability measures coincide. Thus, using \Cref{lem:from-expected-rewards-to-evts} it follows that
\begin{equation}\label{eq:visiting-time-equals-exp-rewards}
    \Exp^\MC[vt_s] = \Exp^\MDP[tr_{rew}]\ .
\end{equation}
Therefore, we can calculate the $\ell$EVT and uEVT of $s\in \IMC$ under the UMC semantic by finding the minimum and maximum expected total rewards for the BMDP $\BMDP_s$, as
\begin{align}
    \ExpLower^\IMC[vt_s] 
    & = \inf_{\MC\models_U}\Exp^\MC[vt_s] && \text{(see \Cref{def:EVTs_of_IDTMCs})}\notag\\
    & = \inf_{\MDP\models\BMDP} \Exp^\MDP[tr_{rew}] && \text{(using \Cref{eq:visiting-time-equals-exp-rewards})}\notag\\
    & = \ExpLower^{\BMDP_s}[tr_{rew}] && \text{(see \Cref{def:expected-total-rewards-of-bmpds})}\ .\notag
\end{align}
The same can be shown for the uEVT of $s$, using the definitions of $\ExpLower^\IMC[vt_s]$ and $\ExpLower^{\BMDP_s}[tr_{rew}]$.

In \cite{Tewari_Bounded_Parameter_MDP}, the authors prove that the minimum and maximum expected total rewards of any BMDP $\BMDP$ can be calculated in PTIME. Hence, we can calculate the $\ell$EVTs and uEVTs of IDTMCs $\IMC$ in PTIME, using the minimum and expected total rewards of the corresponding BMDP $\BMDP$.

We now show that the decision problem of $\ell$EVTs and uEVTs is P-complete, by reducing it to computing minimum and maximum reachability probabilities, which is P-complete. As both $\ell$EVTs and uEVTs as well as minimum and maximum reachability probabilities coincide for the two semantics, it suffices to prove the following reduction for the UMC semantic.

First, we introduce the decision problem of $\ell$EVTs and uEVTs, as well as optimal reachabilities.
\begin{definition}[Decision Problem of $\ell$EVTs and uEVTs]
    Given an IDTMC $\IMC$, a state $s\in S$, and $\underline{x} \in \RRgz$, decide whether $\ExpLower^\IMC[vt_s] = \underline{x}$ 
    
    Similarly, given $\overline{x} \in \RRgz$, decide whether $\ExpUpper^\IMC[vt_s] = \overline{x}$.
\end{definition}

\begin{definition}[Decision Problem of Optimal Reachability Probabilities]
    Given an IDTMC $\IMC$, a state $s\in S$, and $\underline{x} \in \RRgz$, decide whether $\Prlower^\IMC[vt_s] = \underline{x}$ 
    
    Similarly, given $\overline{x} \in \RRgz$, decide whether $\Prupper^\IMC[vt_s] = \overline{x}$.
\end{definition}

Thus, we can now formulate the following theorem.
\begin{theorem}\label{th:u/lEVTs_P-complete}
    The decision problem of $\ell$EVTs and uEVTs is P-complete.
\end{theorem}

\emph{Reduction.} For the rest of this section, we fix a set of states $T\subseteq S^\IMC$ and a state $s \notin S^\IMC$ for the IDTMC $\IMC$ which we fixed at the beginning of this chapter.
We then construct the IDTMC $\IMC_{red} = (S^\IMC \cup \{s,\bot\}, \PFlower^{\IMC_{red}},\PFupper^{\IMC_{red}},\Initlower^\IMC,\Initupper^\IMC)$. Furthermore, we define $\PFupper^{\IMC_{red}}(s_i,s_j) = \PFupper^\IMC(s_i,s_j)$ and $\PFlower^{\IMC_{red}}(s_i,s_j) = \PFlower^\IMC(s_i,s_j)$ for all $s_i\in S^\IMC \backslash T$ and $s_j\in S^\IMC$. Moreover, $\PFupper^{\IMC_{red}}(s_i,s_j) = \PFlower^{\IMC_{red}}(s_i,s_j) = 0$ and $\PFupper^{\IMC_{red}}(s_i,s) = \PFlower^{\IMC_{red}}(s_i,s) = 1$ for $s_i\in T$ and $s_j\in S^\IMC$. Finally, $\PFupper^{\IMC_{red}}(s,\bot) = \PFlower^{\IMC_{red}}(s,\bot) = 1$ and $\PFupper^{\IMC_{red}}(\bot,\bot) = \PFlower^{\IMC_{red}}(\bot,\bot) = 1$.

In other words, this IDTMC $\IMC_{red}$ is identical to $\IMC$, except that all states $t\in T$ can only transition to the state $s$ which in turn can only transition to the absorbing state $\bot$. Notice that $s$ has the recurrence probability of $Pr_s^\MC(\cCircle\lozenge s) = 0$ for all DTMCs $\MC \models_U \IMC_{red}$, as the absorbing state $\bot$ is the only direct successor of $s$.

Finally, we can now prove \Cref{th:reduction_computes_reach_prob}.
\begin{proof}
    We simply need to combine the results of the lemmata. According to \Cref{lem:equal_reach_prob_for_reduction} $Pr^{\MC_{max}}(\lozenge T) = Pr^{\MC_{red}}(\lozenge T)$. Furthermore, using \Cref{lem:equal_reach_prob_for_state_extension}, $Pr^{\MC_{red}}(\lozenge T) = Pr^{\MC_{red}}(\lozenge s)$. Finally, \Cref{lem:EVT_equal_to_reach_prob_for_reduction} shows that $Pr^{\MC_{red}}(\lozenge s) = \Exp^{\MC_{red}}[vt_s]$. Thus, combining these result yields $Pr^{\MC_{max}}(\lozenge T) = \Exp^{\MC_{red}}[vt_s]$. By definition $\Exp^{\MC_{red}}[vt_s] = \ExpUpper^{\IMC_{red}}[vt_s]$ and $Pr^{\MC_{max}}(\lozenge T) = \Prupper^\IMC(\lozenge T)$. \end{proof}

Conversely, the same holds for the $\ell$EVTs by defining $\Exp^{\MC_{red}}[vt_s] = \ExpLower^{\IMC_{red}}[vt_s]$ and $Pr^{\MC_{min}}(\lozenge T) = \Prlower^\IMC(\lozenge T)$ for a DTMC $\MC_{min} \models_U \IMC$.

Thus, as we can calculate the minimum and maximum reachability probabilities of any given $T\subseteq S^\IMC$ using $\ell$EVTs and uEVTs, we can decide the problem of optimal reachability probabilities for $T$ by using $\underline{x} = \ExpLower^\IMC_{red}[vt_s]$ and $\overline{x} = \ExpUpper^\IMC_{red}[vt_s]$. As the problem of optimal reachability probabilities is P-complete, see \cite{complexity_model_checking_interval-valued_DTMCs}, it thus follows that the decision problem of $\ell$EVTs and uEVTs is P-complete as well. Hence, \Cref{th:u/lEVTs_P-complete} is proven.
