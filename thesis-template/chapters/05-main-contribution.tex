\chapter{\textit{EVTs in IDTMCs}}
\label{ch:EVTs_in_IDTMCs}

We continue by defining \textit{expected visiting times} (EVTs) for IDTMCs. As we discussed in the previous section, IDTMCs can be seen as a set of DTMCs that satisfy the IDTMC according to the given semantic. These different DTMCs can have varying EVTs for a given state $s$ in the IDTMC, thus we are interested in the upper and lower bound for the EVTs of $s$, i.e., in the highest and lowest EVT of $s$ among the satisfying DTMCs.
\begin{definition}[EVTs of IDTMCs]\label{def:EVTs_of_IDTMCs}
    For a IDTMC $\IMC$, we define the \textit{upper expected visiting time} (uEVT) $\ExpUpper^\IMC[vt_s]$ and \textit{lower expected visiting time} ($\ell$EVT) $\ExpLower^\IMC[vt_s]$ for all states $s\in S^\IMC$, as follows for the UMC semantic
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s] = \inf_{\MC\models_U\IMC} \Exp^\MC[vt_s]$
        \item $\ExpUpper^\IMC[vt_s] = \sup_{\MC\models_U\IMC} \Exp^\MC[vt_s]$
    \end{enumerate}
    and the IMDP semantic
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s]
        = \inf_{\MC \models_I \IMC} \Exp^\MC[vt_T]$
        \item $\ExpUpper^\IMC[vt_s]
        = \sup_{\MC \models_I \IMC} \Exp^\MC[vt_T]\ ,$
    \end{enumerate}
    where $T = \tau^{-1}(s)$.
\end{definition}

\section{Calculating Expected Visiting Times of IDTMCs}\label{sec:calculating_expected_visiting_times_of_IDTMCs}
We will now have a look at how the \textit{expected visiting times} for IDTMCs can be calculated.

The most intuitive upper and lower bounds for the uEVTs and lEVTs of IDTMCs can be derived by using \Cref{eq:EVT_DTMC_iterative_characterization} and the upper and lower transition probability functions and initial distributions defined in \Cref{def:IMC}.

\begin{theorem}[Upper and Lower Bounds for uEVTs and lEVTs]\label{th:loose_bounds_for_evts}
    For any IDTMC $\IMC$, the uEVT is bounded from above by 
    \[\ExpUpper[vt_s] \leq \Exp_{upp}[vt_s] \coloneq \Initupper(s)+\sum_{t\in S_{tr}}\PFupper(t,s)\cdot \Exp_{upp}[vt_t]\]
    and the $\ell$EVT is bounded from below by 
    \[\ExpLower[vt_s] \geq \Exp_{low}[vt_s] = \Initlower(s)+\sum_{t\in S_{tr}}\PFlower(t,s)\cdot \Exp_{low}[vt_t]\]
    for all $s\in S^\IMC$.
\end{theorem}

In the following, we will therefore take a look at how we can characterize the precise values of the uEVTs and $\ell$EVTs.

First, we provide some characterizations under the UMC semantic, as our definition of transient and recurrent states in IDTMCs is limited to the UMC semantic, see \Cref{def:transient_recurrent_states_IDTMC}. 
\begin{theorem}\label{th:Expupper_ExpLower_infty}
    For a given state $s$ in an IDTMC $\IMC$, $\ExpUpper[vt_s] = \infty$ if and only if $s\in S_{re}^\IMC$.
    Moreover, $\ExpUpper[vt_s] = \ExpLower[vt_s] = \infty$ if and only if $s\in S^\IMC_{\forall re}$.
\end{theorem}

\begin{proof}
    By our \Cref{assmp:finiteness_IDTMCs}, there exists a DTMC $\MC \models_U \IMC$ such that $s$ is reachable in $\MC$, i.e., $Pr^\MC(\lozenge s)>0$ and $\Prupper^\IMC(\lozenge s) > 0$. Moreover, as $s\in S_{re}^\IMC$, there exists a DTMC $\MC_{re}\models_U \IMC$ such that $Pr^{\MC_{re}}_s(\cCircle\lozenge s) = 1$, see \Cref{def:transient_recurrent_states_IDTMC}, thus $\Prupper^{\IMC}_{s}(\cCircle\lozenge s) = 1$. Using \Cref{th:existence_of_dtmc_with_universal_max_reach}, there exists a DTMC $\MC_{+re}\models\IMC$ such that $Pr^{\MC_{+re}}(\lozenge s)>0$ and $Pr^{\MC_{+re}}_{N(s)}(\lozenge s) = 1$, as $\Prupper^\IMC(\lozenge s) > 0$ and $\Prupper^{\IMC}_{s}(\cCircle\lozenge s) = 1$. Therefore, clearly $s\in S^{\MC_{+re}}_{re}$, see \Cref{def:transient_and_recurrent_states} and using \Cref{lem:EVTs_of_recurrent_states}, we can thus conclude that $\Exp^{\MC_{+re}}[vt_s] = \infty$ and thus $\ExpUpper^\IMC[vt_s] =\infty$.
    
    Conversely, if $\ExpUpper^\IMC[vt_s] =\infty$, then there exists a DTMC $\MC \models \IMC$ such that $\Exp^\MC[vt_s] = \infty$, i.e. $s\in S^\MC_{re}$, see \Cref{lem:EVTs_of_recurrent_states}. Thus, it follows directly that $s\in S^\IMC_{re}$, see \Cref{def:transient_recurrent_states_IDTMC}.

    Moreover, $\ExpUpper[vt_s] = \ExpLower[vt_s] = \infty$ if and only if $s\in S^\IMC_{\forall re}$ as $Pr^{\MC}(\lozenge s) > 0$ and $Pr^{\MC}_s(\cCircle\lozenge s) > 0$ for all $\MC \models_U \IMC$, by the definition of $S_{\forall re}$. Thus, using \Cref{lem:EVTs_of_recurrent_states} we can directly conclude the proof.
\end{proof}

\emph{\textbf{Intermediate States.}} Using the notion of finite paths, we introduce \textit{intermediate states}.

\begin{definition}[Intermediate States]\label{def:intermediate_state}
    Given a DTMC $\MC$, a set of states $T$, and a state $s \in S$, then $u\in S$ is \textit{intermediate} if and only if there exists path $\pi \in Paths^\MC_{fin}$ with $|\pi| = n$, such  that $\pi[0]\in T$, $\pi[n] = s$, and $\pi[k] = u$ for some $k\in \underline{n}$. 
\end{definition}
In other words, there exists a finite path from some $t\in T$ to $s$ in $\MC$ that visits $u$.
We denote the set of these intermediate states as $I^\MC_{T,s}$ for a set of states $T\in S$ and state $s\in S$. Moreover, if $T$ is a singleton set, i.e., $T =\{t\}$, then we simply denote the set of intermediate states as $I^\MC_{t,s}$.

\begin{theorem}\label{th:max_reach_prob_for_intermediate_states}
    Given an IDTMC $\IMC$, states $t,s\in S^\IMC$, and a DTMC $\MC$ with $\MC\models_U \IMC$.

    If $\MC$ has the maximum reachability probability from $t$ to $s$, i.e., $Pr^{\MC}_{t}(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$, then for all $u\in I^\MC_{t,s}$:
    \[Pr^{\MC}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)\]

    Conversely, if $\MC$ has the minimum reachability probability from $t$ to $s$, i.e., $Pr^{\MC}_{t}(\lozenge s) = \Prlower^{\IMC}_{t}(\lozenge s)$, then for all $u\in I^\MC_{t,s}$:
    \[Pr^{\MC}_{u}(\lozenge s) = \Prlower^{\IMC}_{u}(\lozenge s)\]
\end{theorem}

In other words, if the reachability probability, under the UMC semantic, from $t$ to $s$ is optimal, then the reachability probability of all intermediate states is optimal as well.

Intuitively, this lemma states that using two DTMCs $\MC_T$ and $\MC_t$ which have optimal reachability probabilities from $T$ and $t$ to $s$, respectively, we can construct a DTMC $\MC_{cmb}$ that has optimal reachability from $T\cup\{t\}$ to $s$.

\begin{proof}
    First, as $\MC_t,\MC_T \models_U \IMC$ it follows from the construction that also $\MC_{cmb} \models \IMC$. Next, as for each $w\in S^\IMC$ either $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{t}}(w,v)$ or $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{T}}(w,v)$ for all $v\in S^\IMC$ it follows that $\PF^{\MC_{cmb}}$ is well-defined, i.e., a discrete probability distribution, see \Cref{def:discrete_probability_distribution}. 

    As $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{T}}(w,v)$ for all $w \in I^{\MC_T}_{T,s}$, $\Prupper^{\IMC}_{u}(\lozenge s) = Pr^{\MC_{cmb}}_u(\lozenge s)$ for all $u\in T$, see \Cref{lem:recursive_formula_for_reachability_prob}.

    Thus, we need to show that $Pr^{\MC_{cmb}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$. As we know that $Pr^{\MC_{t}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$, we will prove that $Pr^{\MC_{cmb}}_t(\lozenge s) = Pr^{\MC_{t}}_t(\lozenge s)$.
    Notice, that if $I^{\MC_t}_{t,s} \cap I^{\MC_T}_{T,s} = \emptyset$, then the proof concludes using the same argument as for $u\in T$. Furthermore, if $I^{\MC_t}_{t,s}\subseteq I^{\MC_T}_{T,s}$, then the proof concludes directly using \Cref{th:max_reach_prob_for_intermediate_states}.

    Recall, that $Pr^{\MC_{cmb}}_{t}(\lozenge s) = \sum_{u\in N(t)\cap I^{\MC_{cmb}}_{t,s}}\PF^{\MC_{cmb}}(t,u)\cdot Pr_u^{\MC_{cmb}}(\lozenge s)$, see \Cref{lem:recursive_formula_for_reachability_prob}. For all $u\in I^{\MC_t}_{t,s}\backslash I^{\MC_T}_{T,s}$, $\PF^{\MC_{cmb}}(u,w) = \PF^{\MC_{t}}(u,w)$ and for all $u\in I^{\MC_T}_{T,s} \cap I^{\MC_t}_{t,s}$, $Pr^{\MC_{cmb}}_u(\lozenge s) = Pr^{\MC_t}_u(\lozenge s)$, see \Cref{th:two_DTMCs_compatible_for_max_reach}. From this, we can conclude that $Pr^{\MC_{t}}_{t}(\lozenge s) = Pr^{\MC_{cmb}}_{t}(\lozenge s)$.
    Thus, we have shown that $Pr_u^{\MC_{cmb}}(\lozenge s) = \Prupper_u^{\MC_{cmb}}(\lozenge s)$ for all $u\in T \cup \{t\}$.

    Conversely, the same holds for minimum reachability probabilities using this reasoning, given that $Pr_u^{\MC_T}(\lozenge s) = \Prlower_u^\IMC(\lozenge s)$ for all $u\in T$ and $Pr_t^{\MC_t}(\lozenge s) = \Prlower_t^\IMC(\lozenge s)$.
\end{proof}

\subsection{UMC semantic}\label{sec:UMC_semantic}
In this section we examine how the precise values of the uEVTs and lEVTs can be calculated for IDTMCs under the UMC semantic.

As defined, there can be multiple DTMCs $\MC$ that satisfy an IDTMC $\IMC$ with differing transition probability functions $\PF^{\MC}$ and initial distributions $\iota_{init}^{\MC}$. \[\Exp[vt_s] = \operatorname{opt}_{\MC\models\IMC} \left(\iota_{init}(s) + \sum_{t\in S_{tr}}\PF^\MC(t,s)\cdot\Exp[vt_t]\right), \qquad \operatorname{opt}\in\{\min,\max\}. \]

% 4. Conclude proof
Finally, we can now prove \Cref{th:reduction_computes_reach_prob}.
\begin{proof}
    We simply need to combine the results of the lemmata. According to \Cref{lem:equal_reach_prob_for_reduction} $Pr^{\MC_{max}}(\lozenge T) = Pr^{\MC_{red}}(\lozenge T)$. Furthermore, using \Cref{lem:equal_reach_prob_for_state_extension}, $Pr^{\MC_{red}}(\lozenge T) = Pr^{\MC_{red}}(\lozenge s)$. Finally, \Cref{lem:EVT_equal_to_reach_prob_for_reduction} shows that $Pr^{\MC_{red}}(\lozenge s) = \Exp^{\MC_{red}}[vt_s]$. Thus, combining these results $Pr^{\MC_{max}}(\lozenge T) = \Exp^{\MC_{red}}[vt_s]$. By definition $\Exp^{\MC_{red}}[vt_s] = \ExpUpper^{\IMC_{red}}[vt_s]$ and $Pr^{\MC_{max}}(\lozenge T) = \Prupper^\IMC(\lozenge T)$. Therefore, the given maximum reachability probability $\Prupper^\IMC(\lozenge T)$ of any IDTMC $\IMC$ can be determined through calculating uEVTs.
\end{proof}

Conversely, the same holds using this reasoning for the $\ell$EVTs.

Thus, as we can calculate the reachabilty minimum and maximum reachability probabilities of any given $T\subseteq S^\IMC$ using $\ell$EVTs and uEVTs, we can decide the problem of optimal reachability probabilities using $\underline{x} = \ExpLower^\IMC_{red}[vt_s]$ and $\overline{x} = \ExpUpper^\IMC_{red}[vt_s]$. As the decision problem of optimal reachability probabilities is P-complete, it thus follows that the decision problem of $\ell$EVTs and uEVTs is P-complete as well.
