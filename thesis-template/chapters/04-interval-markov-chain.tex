\chapter{\textit{Interval Markov Chains}}
\label{ch:Interval Markov Chains}

The focus of this thesis are expected visiting times in Interval Markov Chains, which we will introduce in this chapter. In the following sections we will mainly discuss how the expected visiting times can be calculated and shortly examine stationary distributions of Interval Markov Chains.
 
We start by introducing concepts which we will use in the following sections.
\begin{definition}[Interval Markov Chain]\label{def:IMC}
    An interval discrete-time Markov chain (IDTMC) is a tuple
    $\IMC=(S,\PFlower,\PFupper,\Initlower,\Initupper)$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S$ is a countable, nonempty set of states,
        \item $\PFlower: S \times S \rightarrow [0,1]$ is the lower transition probability function, where $\PFlower(s,s')$ gives the lower bound on the transition probability from state s to s',
        \item $\PFupper: S \times S \rightarrow [0,1]$ is the upper transition probability function, where $\PFupper(s,s')$ gives the upper bound on the transition probability from state s to s',
        \item $\Initlower: S \rightarrow [0,1]$ is the lower bound on the initial distribution, and
        \item $\Initupper: S \rightarrow [0,1]$ is the upper bound on the initial distribution.
    \end{enumerate}
\end{definition}

\begin{definition}[Path of an IDTMC]
    Similarily to DTMCs, an \textit{infinite path} in an IDTMC $\IMC$ is a sequence $\pi = (s_0,s_1,\dots)$ of states with $s_0,s_1,\dots \in S^\IMC$, such that $\PFlower(\pi[k],\pi[k+1])>0$ for all $k\in \NN$.
\end{definition}

In this paper, we will consider two popular semantics: the \textit{once-and-for-all}, also know as the \textit{Uncertain Markov Chain (UMC) semantic}, and the \textit{IMDP semantic}.
We will first define the former, similar to \cite{Semantics_Quant-Analysis,Semantics_Model-Checking_omega-Regular,Semantics_LTL_Model_Checking}. 

\begin{definition}[Once-and-for-all/UMC Semantic]\label{def:once-and-for-all_semantic}
    An arbitrary DTMC $\MC$ satisfies an IDTMC $\IMC$ if and only if $\PFlower^{\IMC}(s,s')\leq\PF^{\MC}(s,s')\leq\PFupper^{\IMC}(s,s')$ and $\Initlower^{\IMC}(s)\leq\iota_{init}^{\MC}(s)\leq\Initupper^{\IMC}(s)$ for all $s,s'\in S^\IMC = S^\MC$. 
    
    We will denote this as $\MC\models_{\text{U}}\IMC$.
\end{definition}
In other words, any DTMC whose initial distribution and transition probabilities lie in the intervals defined by the IDTMC is satisfactory. Moreover, notice that the satisfaying DTMCs have the same state size (amount of states) as the IDTMC.

\end{example}

The second semantic which we will define, is the \textit{IMDP semantic}. There are two common definitions for this semantic. In this thesis, we will use the definition found in \cite{Model-Checking_presence_Uncertainties} and \cite{Semantics_Model-Checking_omega-Regular}.

\begin{definition}[IMDP semantic]\label{def:IMDP_semantic}
    We define $Steps(s)$ as the set of probability functions over $S^\IMC$, i.e. $Steps(s) = \{\mu: S\rightarrow [0,1] \mid \sum_{t\in S} \mu(t) = 1 \text{ and } \PFlower(s,t) \leq \mu(t) \leq \PFupper(s,t) \text{ for all } t\in S\}$ for all $s\in S^\IMC$. Let $last(\pi)$ denote the last state of a finite path $\pi \in Paths^{\IMC}_{fin}$, i.e., if $|\pi| = n$, then $last(\pi) = \pi[n]$.
    
    The choice of a transition function is made by \textit{nature}. A \textit{nature} for an IDTMC $\IMC$ is a function $\eta: Paths^\IMC_{fin} \rightarrow Steps$ that assigns each finite path to a possible probability transition function, i.e., $\eta(\pi)\in Steps(last(\pi))$.

    In other words, for any given finite path $\pi \in Paths_{fin}^\IMC$, the \textit{nature} chooses a probability distribution from $Steps(last(\pi))$ according to which the successor state is then sampled.
    (Still need to define $Paths_{fin}$ for IDTMCs)
\end{definition}

% Intuitively, the IMDP semantic allows for a probability distribution to be chosen at every step in each state within the constraints/bounds set by the IDTMC. In contrast to the UMC semantic, where each state in the satisfying DTMCs have a singular corresponding state in the IDTMC and thus a fixed probability distribution, the IMDP semantic allows for multiple states in the satisfying DTMCs with different fixed probability distributions to be mapped to the same corresponding state in the IDTMC. We can thus imagine an IDTMC under the IMDP semantic as a Markov Decision Process, see \cref{def:MDP}.

As the IMDP semantic allows for a probability distribution to be chosen in each state at every step, we can interpret an IDTMC under the IMDP semantic as a Markov Decision Process (MDP), see \cref{def:MDP}, as the notion of \textit{nature} is identical to the notion of \textit{schedulers} used to resolve the uncertainty in MDPs. 

As defined in \cite{Bibel}.
\begin{definition}[Scheduler]
    Let $\MDP = (S,Act,A,\PF,\iota_{init})$ be a MDP. A \textit{scheduler} for $\MDP$ is a function $\mathfrak{S}: Paths_{fin}^\MDP \rightarrow Act$ such that $\mathfrak{S}(s_0,\dots,s_n)\in A(s_n)$.
\end{definition}

By resolving the uncertainty of an MDP, a scheduler induces a Markov Chain. In \cite{Bibel} (Definition 10.92) these induced Markov Chains are defined as follows

\begin{definition}[Markov Chain of an MDP Induced by a Scheduler]
    For an MDP $\MDP = (S,Act,A,\PF,\iota_{init})$ and a scheduler $\mathfrak{S}$ on $\MDP$, the induced Markov Chain $\MC_{\mathfrak{S}}$ is given by
    \[\MDP_{\mathfrak{S}} = (Paths^\MDP_{inf},\PF_{\mathfrak{S}},\iota_{init})\]
    where for $\sigma = s_0,s_1,\dots,s_n$:
    \[\PF_{\mathfrak{S}}(\sigma,\sigma s_n) = \PF(s_n,\mathfrak{S}(\sigma),s_{n+1})\ .\]
\end{definition}

These induced Markov Chains are precisely the satisfying Markov Chains. Notice, that even if the MDP is finite, any induced Markov Chain is infinite.

The most intuitive reduction from an IDTMC $\IMC$ under the IMDP semantic to an MDP $\MDP$ is to define $S^\MDP = S^\IMC$ and an uncoutable set $\bigcup_{s\in S} A(s) = Act$ of actions where each action $\alpha \in A(s)$ corresponds to a probability distribution in $Steps(s)$. In other words, the actions $\alpha\in A(s)$ in a given state $s\in S$ correspond to the probability distributions that can be chosen in $s$ in any given step. However, as these subsets $A(s)$ can be uncoutable, this reduction is not useful (as no algorithm like value or policy iteration would terminate).

Instead, it is thus argued in \cite{Model-Checking_presence_Uncertainties}, \cite{Semantics_Model-Checking_omega-Regular}, and \cite{complexity_model_checking_interval-valued_DTMCs}, that the local constraints of each state of these MDPs and IDTMCs, see in \cref{def:MDP} $(iv) - (v)$ and \cref{def:IMC} $(ii) - (v)$, can be used to limit the amount of probability distributions that need to be mapped. In the theory of linear programming, this is called a basic feasible solution (BFS). For this reduction, the sets $A(s)$ with $s\in S^\IMC$ are exponential in the size of $|S^\IMC|$.

Important for this thesis are in particular memoryless schedulers. The following definition is taken from \cite{Bibel}.
\begin{definition}[Memoryless Scheduler]
    Let $\MDP$ be a MDP with state space $S$. Any scheduler $\mathfrak{S}$ on $\MDP$ is \textit{memoryless} if and only if for each sequence $(s_0,s_1,\dots,s_n)$ and $(t_0,t_1,\dots,t_m)$ with $s_n = t_m$
    \[\MC(s_0,s_1,\dots,s_n) = \MC(t_0,t_1,\dots,t_m)\ . \]
    Thus, in this case, $\mathfrak{S}$ can be viewed as a function $\mathfrak{S}: S \rightarrow Act$.
\end{definition}

In other words, if a scheduler is memoryless, then it always selects the same action in a given state. Notice, that in regards to the MDP construction in \cite{Model-Checking_presence_Uncertainties} and \cite{Semantics_Model-Checking_omega-Regular}, where the actions represent the chosen probability distribution, this describes precisely the UMC semantic. As a memoryless scheduler chooses the same action in a given state, it chooses the same probability distribution for a given state in the MDP construction.

\begin{definition}[Markov Chain of an MDP Induced by a memoryless Scheduler]
    For an MDP $\MDP = (S,Act,A,\PF,\iota_{init})$ and a scheduler $\mathfrak{S}$ on $\MDP$, the induced Markov Chain $\MC_{\mathfrak{S}}$ is given by
    \[\MDP_{\mathfrak{S}} = (S,\PF_{\mathfrak{S}},\iota_{init})\]
    where for any $s_i,s_j\in S$:
    \[\PF_{\mathfrak{S}}(s_i,s_j) = \PF(s_i,\mathfrak{S}(s_i),s_j)\ .\]
\end{definition}

As $\PFlower(s_i,s_j)\leq\PF(s_i,\mathfrak{S}(s_i),s_j)\leq\PFupper(s_i,s_j)$ by construction, the induced Markov Chain of a memoryless scheduler on the MDP construction satisfies all conditions of \cref{def:once-and-for-all_semantic}.

We use the definition of the quantitative reachability property in IDTMCs provided in \cite{reachability_in_pIMCs_using_constraints}. 
\begin{definition}[Reachability Probability in IDTMCs]\label{def:reachability_probability_IDTMCs}
    Given an IDTMC $\IMC$ and a set of states $T \subseteq S$, then a quantitative reachability property on $\IMC$ is defined as $Pr^\IMC(\lozenge T) \sim p$, where $0\leq p \leq 1$ and $\sim \in \{\leq, <, >, \geq\}$. This quantitative property is satisfied if and only if there exists a DTMC $\MC \models \IMC$ such that $Pr^\MC(\lozenge T) \sim p$.
\end{definition}

\begin{definition}[Transient and Recurrent states in IDTMCs]\label{def:transient_recurrent_states_IDTMC}
    Similar to DTMCs, a state $s\in S^\IMC$ is called recurrent if and only if $Pr^\IMC_s(\cCircle\lozenge s) = 1$, i.e., there exists a satisfying DTMC $\MC\models\IMC$ such that $Pr^\MC_s(\cCircle\lozenge s) = 1$. All states that are not recurrent are called transient.
\end{definition}

\end{definition}

\subsection{Relation between the semantics}\label{sec:relation_between_the_semantics}
Clearly, the IMDP semantic is more expressive than the UMC semantic \cite{Semantics_Quant-Analysis}. More specifically for any given IDTMC $\IMC$, a DTMC $\MC$ that satisfies $\IMC$ according to the UMC semantic, i.e. $\MC \models_U \IMC$, does also satisfy $\IMC$ according to the IMDP semantic, i.e. $\MC \models_I \IMC$ \cite{Semantics_Quant-Analysis}.
\begin{theorem}
    For any IDTMC $\IMC$, if $\MC \models_U \IMC$, then $\MC \models_I \IMC$
\end{theorem}
For this, we provide a slightly more detailed proof of \cite{Semantics_Quant-Analysis} (Remark 1).

\begin{proof}
    If $\MC \models_U \IMC$, then we define $T = S$ and $\tau(s) = s$. Then (i) of \cref{def:IMDP_semantic} is fulfilled as $\tau$ is bijective and $\iota_{init}(s) \in [\Initlower(s),\Initupper(s)]$ for all $s\in S$, as $\MC \models_U \IMC$. Moreover, (ii) is fulfilled as $\PF(s,t)\in[\PFlower(s,t),\PFupper(s,t)]$ for all $s,t\in S$, as $\MC \models_U \IMC$. Thus, we can conclude that $\MC \models_I \IMC$.
\end{proof}

Importantly, the two semantics have the same maximum and minimum reachability probability for any given (initial) state, according to \cite{reachability_in_pIMCs_using_constraints} and \cite{complexity_model_checking_interval-valued_DTMCs}.

\section{Expected Visiting Times in IDTMCs}
We will now define \textit{expected visiting times} (EVTs) for IDTMCs. As we discussed in the previous section, IDTMCs can be seen as a set of DTMCs that satiesfy the IDTMC according to the given semantic. These different DTMCs can have varying EVTs for a given state $s$ in the IDTMC, thus we are interested in the upper and lower bound for the EVTs of $s$, i.e., in the highest and lowest EVT of $s$ among the satisfying DTMCs.
\begin{definition}[EVTs of IDTMCs]
    For a IDTMC $\IMC$, we define the \textit{upper expected visiting time} (uEVT) $\ExpUpper^\IMC[vt_s]$ and \textit{lower expected visiting time} ($\ell$EVT) $\ExpLower^\IMC[vt_s]$ for all states $s\in S^\IMC$, as follows for the UMC semantic
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s] = \operatorname{inf}_{\MC\models_U\IMC} \Exp^\MC[vt_s]$
        \item $\ExpUpper^\IMC[vt_s] = \operatorname{sup}_{\MC\models_U\IMC} \Exp^\MC[vt_s]$
    \end{enumerate}
    and the IMDP semantic
    \begin{enumerate}[label=(\roman*)]
        \item $\ExpLower^\IMC[vt_s] = \operatorname{inf}_{\MC\models_I\IMC} \sum_{t\in T \land \tau^\MC(t) = s}\Exp^\MC[vt_t]$
        \item $\ExpUpper^\IMC[vt_s] = \operatorname{sup}_{\MC\models_I\IMC}\sum_{t\in T \land \tau^\MC(t) = s}\Exp^\MC[vt_t]$.
    \end{enumerate}
\end{definition}

\subsection{Relating Expected Visiting Times and Quantitative Reachability}\label{sec:IMC_Quantitative_Reachability}
Similar to \cref{sec:MC_Quantitative_Reachability}, we will now relate \textit{expected visiting times} to \textit{quantitave reachability}. Moreover, we show that for any IDTMC $\IMC$, there exists a DTMC $\MC\models \IMC$ that has the optimal, i.e., maximum or minimum, reachability probability for all transient states in $\IMC$.

Given the quantitative reachabilities of the DTMCs that satisfy a certain IDTMC $\IMC$, we can calculate the corresponding EVTs of a given state $s$ in $\IMC$ and thus find the lEVTs and uEVTs using the results from \cref{sec:MC_Quantitative_Reachability}. 

As we concluded in \cref{sec:MC_Quantitative_Reachability}, we can calculate the EVT of a given state $s$ in a DTMC $\MC$ via quantitative reachability, using the corresponding DTMC $\MC_{uni}$ and the \cref{eq:EVT_via_quant_reach_sum} $\Exp^\MC[vt_s] = \sum_{t\in\Strin} \frac{Pr^\MC_t(\lozenge s)}{1-Pr^\MC_s(\Circle\lozenge s)} \cdot\iota_{init}(t)$. Thus, the uEVT and $\ell$EVT of a given IDTMC $\IMC$ are given by the largest and smallest ratio of these quantitative reachability probabilites among the satisfying Markov Chains.

\begin{corollary}\label{th:two_DTMCs_compatible_for_max_reach}
    Given an IDTMC $\IMC$, two transient states $t_1, t_2\in S^\IMC$, a (goal) state $s\in S^\IMC$, and two DTMCs $\MC_1,\MC_2$ with $\MC_1,\MC_2\models_U \IMC$ and $Pr^{\MC_1}_{t_1}(\lozenge s) = \Prupper^\IMC_{t_1}(\lozenge s)$ and $Pr^{\MC_2}_{t_2}(\lozenge s) = \Prupper^{\IMC}_{t_2}(\lozenge s)$. In other words the DTMCs have a maximum reachability probability for $t_1$ and $t_2$, respectively.

    For all $u\in I^{\MC_1}_{t_1s} \cap I^{\MC_2}_{t_2s}$, the reachability probabilities $Pr^{\MC_1}_u(\lozenge s) = Pr^{\MC_2}_u(\lozenge s)$ are identical.
\end{corollary}


\begin{proof}
    This corollary follows from \cref{th:max_reach_prob_for_intermediate_states}.
    Using \cref{th:max_reach_prob_for_intermediate_states}, we can conclude that $Pr^{\MC_1}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)$ for all $u \in I^{\MC_1}_{t_1s}$ and $Pr^{\MC_2}_{u}(\lozenge s) = \Prupper^{\IMC}_{u}(\lozenge s)$ for all $u \in I^{\MC_2}_{t_2s}$. Thus, $Pr^{\MC_1}_u(\lozenge s) = Pr^{\MC_2}_u(\lozenge s)$ for all $u\in I^{\MC_1}_{t_1s} \cap I^{\MC_2}_{t_2s}$.
\end{proof}

\begin{lemma}[One-state Extension]\label{lem:one-state_extension}
    Given a set $T \subset S$, a state $t \notin T$, and a (goal) state s, assume that $\MC_T \models_U \IMC$ with $Pr_u^{\MC_T}(\lozenge s) = \Prupper_u^\IMC(\lozenge s)$ for all $u\in T$ and $\MC_t \models_U \IMC$ with $Pr_t^{\MC_t}(\lozenge s) = \Prupper_t^\IMC(\lozenge s)$. Define $\MC_{cmb}\models_U \IMC$ by
    \[\PF^{\MC_{cmb}}(w,v) = 
    \begin{cases*}
        \PF^{\MC_t}(w,v) & if $w\in I^{\MC_t}_{ts}\backslash I^{\MC_T}_{Ts}$ \\
        \PF^{\MC_T}(w,v) & else.
    \end{cases*}\]
    Stated in words, $\MC_{cmb}$ is constructed such that its probability distribution function is equal to that of $\MC_t$ in all states that are only intermediate states for $t$ and not $T$, while it is equal to that of $\MC_T$ in all other states.
    
    Then, $Pr^{\MC_{cmb}}_u(\lozenge s) = \Prupper^{\MC_{cmb}}_{u}(\lozenge s)$ for all $u \in T \cup \{t\}$, i.e., $\MC_{cmb}$ has the maximum reachability probability for all states in $T\cup \{t\}$.
\end{lemma}

\begin{proof}
    As $\PF^{\MC_{cmb}}(w,v) = \PF^{\MC_{T}}(w,v)$ for all $w \in I^{\MC_T}_{Ts}$, clearly $\Prupper^{\IMC}_{u}(\lozenge s) = Pr^{\MC_{cmb}}_u(\lozenge s)$ for all $u\in T$, see \cref{lem:recursive_formula_for_reachability_prob}.

    Thus, we need to show that $Pr^{\MC_{cmb}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$. As we know that $Pr^{\MC_{t}}_t(\lozenge s) = \Prupper^{\IMC}_{t}(\lozenge s)$, we will prove that $Pr^{\MC_{cmb}}_t(\lozenge s) = Pr^{\MC_{t}}_t(\lozenge s)$.
    Notice, that if $I^{\MC_t}_{ts} \cap I^{\MC_T}_{Ts} = \emptyset$, then the proof concludes using the same argument as for $u\in T$. Furthermore, if $I^{\MC_t}_{ts}\subseteq I^{\MC_T}_{Ts}$, then the proof concludes directly using \cref{th:max_reach_prob_for_intermediate_states}.

    Recall, that $Pr^{\MC_{cmb}}_{t}(\lozenge s) = \sum_{u\in N(t)\cap I^{\MC_{cmb}}_{ts}}\PF^{\MC_{cmb}}(t,u)\cdot Pr_u^{\MC_{cmb}}(\lozenge s)$, see \cref{lem:recursive_formula_for_reachability_prob}. For all $u\in I^{\MC_t}_{ts}\backslash I^{\MC_T}_{Ts}$, $\PF^{\MC_{cmb}}(u,w) = \PF^{\MC_{t}}(u,w)$ and for all $u\in I^{\MC_T}_{Ts} \cap I^{\MC_t}_{ts}$, $Pr^{\MC_{cmb}}_u(\lozenge s) = Pr^{\MC_t}_u(\lozenge s)$, see \cref{th:two_DTMCs_compatible_for_max_reach}. From this, we can conclude that $Pr^{\MC_{t}}_{t}(\lozenge s) = Pr^{\MC_{cmb}}_{t}(\lozenge s)$.
    Thus, we have shown that $Pr_u^{\MC_{cmb}}(\lozenge s) = \Prupper_u^{\MC_{cmb}}(\lozenge s)$ for all $u\in T \cup \{t\}$.

    Into only remains to show, that $\MC_{cmb} \models_U \IMC$. As $\MC_t,\MC_T \models_U \IMC$ it clearly follows from the construction that also $\MC_{cmb} \models \IMC$.
\end{proof}

Crucially, \cref{lem:one-state_extension} can be extended to all transient states.

\section{Calculating Expected Visiting Times of IDTMCs}
We will now have a look at how the \textit{expected visiting times} for IDTMCs can be calculated

The most intuitive upper and lower bounds for the uEVTs and lEVTs of IDTMCs can be derived by using \cref{eq:EVT_DTMC_iterative_characterization} and the upper and lower transition probability functions and initial distributions defined in \cref{def:IMC}.

\begin{theorem}[Upper and Lower Bounds for uEVTs and lEVTs]\label{th:loose_bounds_for_evts}
    For any IDTMC $\IMC$, the uEVT is bounded from above by $\ExpUpper[vt_s] \leq \Exp_{upp}[vt_s] = \Initupper(s)+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp_{upp}[vt_t]$ and the $\ell$EVT is bounded from below by $\ExpLower[vt_s] \geq \Exp_{low}[vt_s] = \Initlower(s)+\sum_{t\in S_{tr}}\PFlower(t,s)^\T\cdot \Exp_{low}[vt_t]$ for all $s\in S_{tr}^\IMC$.
\end{theorem}
\begin{proof}
    For any IDTMC $\IMC$ there exists a DTMC $\MC$ that satisfies $\IMC$, i.e., $\MC \models \IMC$, and $\Exp^\MC[vt_s] = \ExpUpper^\IMC[vt_s]$. As $\MC \models \IMC$, it follows that $\PF(s,s')\leq \PFupper(s,s')$ for all $s,s'\in S$.

    Thus, $\Exp^\MC[vt_s] = \iota_{init}(s)+\sum_{t\in S_{tr}}\PF(t,s)^\T\cdot \Exp[vt_t] \leq \Initupper(s)+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp_{upp}[vt_t] = \Exp_{upp}^\IMC[vt_s]$.
    
    The identical argument can be made to prove the lower bound for the $\ell$EVT.

\begin{theorem}
    For a given state $s$ in an IDTMC $\IMC$, $\ExpUpper[vt_s] = \infty$ if and only if $s\in S_{re}^\IMC$.
\end{theorem}

\begin{proof}
    By our \cref{assmp:finiteness_(I)DTMCs}, there exists a DTMC $\MC \models \IMC$ such that $s$ is reachable in $\MC$, i.e., $Pr^\MC(\lozenge s)>0$ and $\Prupper^\IMC(\lozenge s) > 0$. Moreover, as $s\in S_{re}^\IMC$, there exists a DTMC $\MC_{re}\models \IMC$ such that $Pr^{\MC_{re}}_s(\cCircle\lozenge s) = 1$, see \cref{def:transient_recurrent_states_IDTMC}, thus $\Prupper^{\IMC}_{s}(\cCircle\lozenge s) = 1$. Using \cref{th:existence_of_dtmc_with_universal_max_reach}, there exists a DTMC $\MC_{+re}\models\IMC$ such that $Pr^{\MC_{+re}}(\lozenge s)>0$ and $Pr^{\MC_{+re}}_{N(s)}(\lozenge s) = 1$, as $\Prupper^\IMC(\lozenge s) > 0$ and $\Prupper^{\IMC}_{s}(\cCircle\lozenge s) = 1$. Therefore, clearly $s\in S^{\MC_{+re}}_{re}$, see \cref{def:transient_and_recurrent_states} and using \cref{lem:EVTs_of_recurrent_states}, we can thus conclude that $\Exp^{\MC_{+re}}[vt_s] = \infty$ and thus $\ExpUpper^\IMC[vt_s] =\infty$.
    
    Conversely, if $\ExpUpper^\IMC[vt_s] =\infty$, then there exists a DTMC $\MC \models \IMC$ such that $\Exp^\MC[vt_s] = \infty$, i.e. $s\in S^\MC_{re}$, see \cref{lem:EVTs_of_recurrent_states}. Thus, it follows directly that $s\in S^\IMC_{re}$, see \cref{def:transient_recurrent_states_IDTMC}.
\end{proof}

\end{theorem}

\subsection{UMC semantic}
In this section we examine how the precise values of the uEVTs and lEVTs can be calculated for IDTMCs under the UMC semantic.

As defined, there can be multiple DTMCs $\MC_i$ that satisfy an IDTMC $\IMC$ with differing transition probability functions $\PF^{\MC_i}$ and initial distributions $\iota_{init}^{\MC_i}$. To determine the uEVT and $\ell$EVT for a given state $s\in S^\IMC$, we thus have to find the satisfying DTMCs that maximize and minimze \cref{eq:EVT_DTMC_iterative_characterization}, respectively.
In other words, we need to solve the equations
\[\Exp[vt_s] = \operatorname{opt}_{\MC\models\IMC} \left(\iota_{init}(s) + \sum_{t\in S_{tr}}\PF^\MC(t,s)\cdot\Exp[vt_t]\right), \qquad \operatorname{opt}\in\{\min,\max\}. \]

As we concluded, the $\ell$EVT and uEVT of a given state $s$ in an IDTMC, can be calculated by computing the reachability probabilities $\Prupper^\IMC(\lozenge s)$ and $\Prupper^\IMC(\cCircle \lozenge s)$ or $\Prlower^\IMC(\lozenge s)$ and $\Prlower^\IMC(\cCircle \lozenge s)$. \[

        \operatorname{opt} \quad & \sum_{s\in S\backslash T}x_s & & \operatorname{opt}\in\{\max,\min\}\\

    \end{aligned}
\]

Recall, that we concluded in \cref{sec:IMC_Quantitative_Reachability} that the uEVT and $\ell$EVT of a given IDTMC $\IMC$ can be calculated by maximizing or minimizing the term $\sum_{t\in \Strin} \frac{Pr_t(\lozenge s)}{1-Pr_s(\cCircle\lozenge s)} \cdot \iota_{init}(t)$ for all DTMCs $\MC \models \IMC$. Intuitively, we can thus find the lEVTs and uEVTs by maximizing or minimizing the reachability probability from the initial state $t$ to the given state $s$ ($Pr_t(\lozenge s)$) and from $s$ back to itself ($Pr_s(\cCircle\lozenge s)$). As shown in \cite{complexity_model_checking_interval-valued_DTMCs} the problem of finding this supremum or infimum for a given IDTMC is P-complete. Thus, we can derive the two suprema or infima belonging to (at most) two different satisfying DTMCs $\MC_{s_{in}\rightarrow s}$ and $\MC_{s\rightarrow s}$ ($\MC_{s_{in}\rightarrow s},\ \MC_{s\rightarrow s}\models_{I} \IMC$).

As stated in \cref{th:k-step_transition_probability} the $k$-th power of the transition matrik $\tT^k$ contains the probabilities of transitioning from any state $i$ to any other state $j$ in precisely $k$ steps. Thus, the product $\iota_{init}^\T \cdot \tT^k$ characterizes the probability distribution of the current state of the given DTMC in step $k$. We define this product as $\theta_k^\T = \iota_{init}^\T \cdot \tT^k$, where the $i$-th entry of $\theta_k$ characterizes the probability that the current state in step $k$ is $i$. We can also define $\theta_k$ iteratively through $\theta_k^\T = \theta_{k-1}^\T \cdot \tT$

\begin{definition}[Stationary Distributions in DTMCs]
    A distribution $\theta_k^\MC$ in a DTMC $\MC$ is stationary if and only if $\theta_k^\T = \theta_k^\T \cdot \tT^\MC$, i.e. the probability distribution of the current state remains unchanged through future steps. We denote the stationary distribution as $\theta^\MC$, where $\theta^\MC(s)$ denotes the entry of $\theta^\MC$ which corresponds to state $s$.
\end{definition}

\begin{definition}[Stationary Distributions in IDTMCs]
    For a given IDTMC $\IMC$, we define $\Theta = \{\theta^\MC \mid \MC \models \IMC\}$ as the set of stationary distributions. Furthermore, for a given state $s\in S^\IMC$, we define $\Thetalower^\IMC(s)$ and $\Thetaupper^\IMC(s)$ such that $\Thetalower^\IMC(s) = \operatorname{inf}_{\MC\models\IMC}\theta^\MC(s)$ and $\Thetaupper^\IMC(s) = \operatorname{sup}_{\MC\models\IMC}\theta^\MC(s)$.
\end{definition}

In contrast to EVTs, when considering the stationary distribution of a DTMC or IDTMC, we are interested in the recurrent, not transient, states.

We denote stationary distribution of a DTMC $\MC$ restricted to a BSCC $B$ as $\theta^{\MC|_B}$.

As proven in \cite{modeling_and_analysis_stochastic_systems}, the stationary distribution $\theta^\MC$ of the recurrent states of a DTMC $\MC$ is characterized by 
\begin{equation}\label{eq:Stat_Distr_via_BSCC_for_DTMC} 
    \theta^\MC(s) = Pr^\MC(\lozenge B) \cdot \theta^{\MC|_B}(s)\ ,
\end{equation} 
for all $s\in B$. Moreover, in \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens}, it is shown that the stationary distribution within a BSCC can be computed through its EVTs. For this they construct a DTMC $\mathcal{B}\!\Rsh^{\hat{v}}$ for every BSCC $B$ with any arbitrary state $v\in S^{\MC|_B}$ in which all incoming transitions of $v$ are redirected to the fresh absorbing state $\hat{v}$. 

\emph{Construction.} More precisely, $\mathcal{B}\!\Rsh^{\hat{v}} = (S^{\MC|_B} \cup \{\hat{v}\},\PF^{\mathcal{B}\Rsh^{\hat{v}}},\iota_{init}^{\mathcal{B}\Rsh^{\hat{v}}})$, where $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(\hat{v},\hat{v}) = 1$, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,\hat{v}) = \PF(s,v)$ and $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,v) = 0$ for all $s\in S^{\MC|_B}$, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,t) = \PF(s,t)$ for all $s\in S$, $t\in S\backslash\{v\}$, and $\iota_{init}^{\mathcal{B}\Rsh^{\hat{v}}}(v) = 1$. 

The stationary distribution within a DTMC is then given by
\begin{equation}\label{eq:Stat_Distr_of_BSCC}
    \theta^{\MC|_B}(s) = \frac{\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_s]}{\sum_{t\in S}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]}\ ,
\end{equation}
as proven in \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens}.
