We define stationary distributions, also called long-run distributions, as done in \cite{Bibel}.
\begin{definition}[Stationary Distributions in DTMCs]
    Given a finite DTMC $\MC = (S,\PF,\iota_{init})$ be a finite Markov chain and $s\in S$. The \textit{long-run average probability} $\theta^\MC(s)$ is given by
    \[ \theta^\MC(s) = \lim_{n\rightarrow \infty} \frac{1}{n} \cdot \sum_{i=1}^{n}Pr(\lozenge^{=k}s)\ .\]
    % The function $t \mapsto \theta(s,t)$ is the \textit{stationary distribution} for starting state $s$. The stationary distribution for $\MC$ is given by
    % \[ \Theta^\MC(t) = \sum_{s\in S} \iota_{init}(s) \cdot \theta(s,t)\ .\]
\end{definition}
In other words, the stationary distribution $\Theta^\MC(t)$ describes the probability that $\MC$ is in state $t$ after $k$ steps, as $k$ tends towards infinity. Whenever the DTMC $\MC$ is evident from the context, we omit the superscript $\MC$ from $\theta^\MC$.

Similarly, we now define stationary distributions for IDTMCs. First, recall that the DTMCs $\MC_{\eta} \models_I \IMC$ that are induced by some nature $\eta$ can be infinite, i.e., $|\MC_{\eta}| = \infty$, see \Cref{def:DTMC_induced_by_nature}. Intuitively, as every state $s\in S^{\MC_{eta}}$ in these DTMCs is only visited once, $\theta(s) = 0$. In other words, there exists no stationary distribution for these states.

\begin{definition}[Stationary Distributions in IDTMCs]
    For a given IDTMC $\IMC$ and state $s\in S^\IMC$, we define 
    \[\Theta(s) = \{\theta(s)^\MC \mid \MC \models_U \IMC\}\]
    as the set of stationary distributions of $s$ on $\IMC$. 
    \\Furthermore, we define $\Thetalower^\IMC(s)$ and $\Thetaupper^\IMC(s)$ such that 
    \[\Thetalower^\IMC(s) = \operatorname{inf}_{\MC\models_U\IMC}\theta^\MC(s) \qquad \text{and} \qquad \Thetaupper^\IMC(s) = \operatorname{sup}_{\MC\models_U\IMC}\theta^\MC(s)\ .\]  
    %\[\]
\end{definition}
In other words, $\Thetalower^\IMC(s)$ and $\Thetaupper^\IMC(s)$ describe the minimum and maximum stationary distribution of $s$ among the DTMCs $\MC \models_U \IMC$ that satisfy $\IMC$ under the UMC semantic.
Whenever the DTMC $\MC$ is evident from the context, we omit the superscript $\IMC$ from $\Thetalower^\IMC$ and $\Thetaupper^\IMC$.


Moreover, BSCCs are defined as follows for IDTMCs.
\begin{definition}[BSCCs for IDTMCs]
    Let $\IMC$ be an IDTMC, then $B^\IMC \subseteq S^\IMC$ is a BSCC, if there exists at least one DTMC $\MC \models_U \IMC$ such that $B^\IMC$ is a BSCC in the DTMC $\MC$, see \Cref{def:(B)SCC}.
\end{definition}
Whenever the DTMC $\MC$ is evident from the context, we omit the superscript $\IMC$ from $B^\IMC$.

Recall, that the recurrent states of a DTMC $\MC$ are precisely the states contained in the BSCCs of $\MC$, see \Cref{sec:DTMCs}. Thus, by collapsing each BSCCS of $\MC$ into a single state, the stationary distribution $\theta^\MC$ of the recurrent states of a DTMC $\MC$ is characterized by 
\begin{equation}\label{eq:Stat_Distr_via_BSCC_for_DTMC} 
    \theta^\MC(s) = Pr^\MC(\lozenge B) \cdot \theta^{\MC|_B}(s)\ ,
\end{equation} 
for all $s\in B$, as proven in \cite{modeling_and_analysis_stochastic_systems}, where $\theta^{\MC|_B}$ denotes the stationary distribution of $\MC$ restricted to $B$. Moreover, in \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens}, it is shown that the stationary distribution within a BSCC can be computed through its EVTs. For this they construct a DTMC $\mathcal{B}\!\Rsh^{\hat{v}}$ for every BSCC $B$ with any arbitrary state $v\in S^{\MC|_B}$ in which all incoming transitions of $v$ are redirected to the fresh absorbing state $\hat{v}$. As $v$ is arbitrary, we fix $v$ and $\hat{v}$ for the remainder of this chapter.

\emph{Construction.} More precisely, $\mathcal{B}\!\Rsh^{\hat{v}} = (S^{B} \cup \{\hat{v}\},\PF^{\mathcal{B}\Rsh^{\hat{v}}},\iota_{init}^{\mathcal{B}\Rsh^{\hat{v}}})$, where $S^B$ denotes the states that are contained in $B$. Moreover, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(\hat{v},\hat{v}) = 1$, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,\hat{v}) = \PF^\MC(s,v)$, and $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,v) = 0$ for all $s\in S^{\MC|_B}$. Lastly, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,t) = \PF^\MC(s,t)$ for all $s\in S$, $t\in S\backslash\{v\}$, and $\iota_{init}^{\mathcal{B}\Rsh^{\hat{v}}}(v) = 1$. 

The stationary distribution within a DTMC is then given by
\begin{equation}\label{eq:Stat_Distr_of_BSCC}
    \theta^{\MC|_B}(s) = \frac{\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_s]}{\sum_{t\in S}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]}\ ,
\end{equation}
as proven in \cite{Computing_EVTs_Stationary_Distr_in_MCs_Mertens}.

Similarly to the DTMC $\mathcal{B}\!\Rsh^{\hat{v}}$, we now define the IDTMC $\mathcal{I}\!\Rsh^{\hat{v}}$ as follows.

\emph{Construction.} $\mathcal{I}\!\Rsh^{\hat{v}} = (S^{B} \cup \{\hat{v}\},\PFlower^{\mathcal{B}\Rsh^{\hat{v}}},\PFupper^{\mathcal{B}\Rsh^{\hat{v}}},\Initlower^{\mathcal{B}\Rsh^{\hat{v}}},\Initupper^{\mathcal{B}\Rsh^{\hat{v}}})$, where $S^B$ denotes the states that are contained in $B$. Moreover, $\PFupper^{\mathcal{B}\Rsh^{\hat{v}}}(\hat{v},\hat{v}) = \PFlower^{\mathcal{B}\Rsh^{\hat{v}}}(\hat{v},\hat{v}) = 1$, $\PFupper^{\mathcal{B}\Rsh^{\hat{v}}}(s,\hat{v}) = \PFupper^\IMC(s,v)$, $\PFlower^{\mathcal{B}\Rsh^{\hat{v}}}(s,\hat{v}) = \PFlower^\IMC(s,v)$, and $\PFupper^{\mathcal{B}\Rsh^{\hat{v}}}(s,v) = \PFlower^{\mathcal{B}\Rsh^{\hat{v}}}(s,v) = 0$ for all $s\in S^{\MC|_B}$. Lastly, $\PF^{\mathcal{B}\Rsh^{\hat{v}}}(s,t) = \PF(s,t)$ for all $s\in S$, $t\in S\backslash\{v\}$, and $\iota_{init}^{\mathcal{B}\Rsh^{\hat{v}}}(v) = 1$. 

Intuitively, as $\MC \models_U \IMC$ it follows that, by construction, $\mathcal{B}\!\Rsh^{\hat{v}} \models_U \mathcal{I}\!\Rsh^{\hat{v}}$.

\begin{assumption}\label{assmp:graph_preserving}
    In the rest of this Chapter, we assume that all IDTMCs $\IMC$ are graph preserving, meaning that there exists no lower transition probability function such that $\PFlower^\IMC(s,t) = 0$ for any $s,t\in S^\IMC$. This implies that there exists no $\MC \models_U \IMC$ such that some $s\in S^\IMC$ is not reachable in $\MC$.
\end{assumption}

Given this assumption, we can prove the following theorem.
\begin{theorem}
    Given an IDTMC $\IMC$, a state $s\in S_{re}^\IMC$, and any pair of BSCCs $B^{\MC},B'^{\MC'} \subseteq S^\IMC$ with $B \neq B'$, it holds that $s\in B$ implies $s\notin B'$.
\end{theorem}
In other words, every state $S_{re}^\IMC$ corresponds to precisely one BSCC in $\IMC$.

\begin{proof}
    Assume that for some IDTMC $\IMC$ there exists a state $s$, two BSCCs $B^{\MC}, B'^{\MC'}$ with $B^\neq B'^{\MC'}$, and $\MC, \MC'\models \IMC$, such that $s\in B^{\MC}$ and $s\in B'^{\MC'}$. As $B^{\MC}\neq B'^{\MC'}$, there exists at least one state $t\in S^\IMC$ that is only contained in either $B^{\MC}$ or $B'^{\MC'}$, i.e., $t\in (B\backslash B') \cup (B'\backslash B)$. 
    
    Without loss of generality, let us assume $t\in B^\MC$. Hence, by \Cref{def:(B)SCC}, there exists a path from $s$ to $t$ in $\MC$. However, as noted the graph relations for all satisfying DTMCs of graph preserving IDTMCs are identical. Thus, there must also exist a path from $s$ to $t$ in $\MC'$. However, $t\notin B'^{\MC'}$, which contradicts \Cref{def:(B)SCC}.
\end{proof}

Thus, we can conclude for graph preserving IDTMCs that any state $s\in S^\IMC$ belongs only to a singular BSCC $B$, i.e., the BSCCs for all satisfying DTMCs are identical. This allows us, to optimize the reachability of $B$ ($Pr^\MC(\lozenge B)$) and the stationary distribution restricted to $B$ ($\theta^{\MC|_B}$) of \Cref{eq:Stat_Distr_via_BSCC_for_DTMC} independently as shown in the following.

Therefore, we can intuitively calculate the stationary distributions for a state $s\in \IMC$ using the following theorem.
\begin{theorem}\label{th:max_min_stat_distribution}
    Given an IDTMC $\IMC$, a state $s$, and the corresponding BSCC $B^\IMC$ with $s\in B^\IMC$, we can calculate the minimum and maximum stationary distribution using
    \[\Thetalower^\IMC(s) = \Prlower^\IMC(\lozenge B) \cdot \frac{\ExpLower^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s]}{\ExpLower^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s] + \sum_{t\in S\backslash\{s\}}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]}\]
    and 
    \[\Thetaupper^\IMC(s) = \Prupper^\IMC(\lozenge B) \cdot \frac{\ExpUpper^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s]}{\ExpUpper^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s] + \sum_{t\in S\backslash\{s\}}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]}\]
    similar to \Cref{eq:Stat_Distr_via_BSCC_for_DTMC} using \Cref{eq:Stat_Distr_of_BSCC}.
\end{theorem}

We provide a short proof by contradiction of this theorem
\begin{proof}
    Let us assume that there exists a DTMC $\MC \models_U \IMC$ such that $\Thetaupper^\IMC(s) < \theta^\MC(s)$. Using \Cref{th:max_min_stat_distribution} and \Cref{eq:Stat_Distr_via_BSCC_for_DTMC}, we can thus conclude that either
    \[Pr^\MC(\lozenge B) > \Prupper^{\IMC}(\lozenge B) \qquad \text{or} \qquad \frac{\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_s]}{\sum_{t\in S}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]} > \frac{\ExpUpper^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s]}{\ExpUpper^{\mathcal{I}\Rsh^{\hat{v}}}[vt_s] + \sum_{t\in S\backslash\{s\}}\Exp^{\mathcal{B}\Rsh^{\hat{v}}}[vt_t]}\ .\]
    However, as $\MC \models_U \IMC$ and $\mathcal{B}\!\Rsh^{\hat{v}} \models_U \mathcal{I}\!\Rsh^{\hat{v}}$, by the definition of $\Prupper^\IMC$ and $\ExpUpper^{\mathcal{I}\Rsh^{\hat{v}}}$ this leads to a contradiciton.
    The proof for $\Thetalower^\IMC(s)$ follows analogously by inverting the inequalities.
\end{proof}
 As $s$ belongs to a singular BSCC $B$, $\mathcal{B}\!\Rsh^{\hat{v}}$ is necessarily constructed from $B$.
