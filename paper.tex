\begin{definition}[Gershgorin circle theorem]\label{def:gershgorin_theorem}
    Let $\tT\in \CC^{n\times n}$ be a square complex matrix, then for every eigenvalue $\lambda$ of $\tT$ it is true that 
    \[|\lambda-(\tT)_{ii}|\leq\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|\ ,\] 
    for any $i\in \underline{n}$ and where $|\cdot|$ denotes the absolute value, i.e. magnitude.\\
    In other words, all eigenvalues of $\tT$ lie on a disk in the complex plane centered at $(\tT)_{ii}$ with radius $\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|$.
\end{definition}
Matrix invertible if 1 not eigenvalue

\begin{definition}[Row-Stochastic Matrix]\label{def:row-stoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-stochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} = 1$ for all rows $i \in \underline{n}$.
\end{definition}
When considering each row as a function $f_i(j)=(\tT)_{ij}$, these functions are discrete probability distributions.

\begin{definition}[(Strictly) Row-Substochastic Matrix]\label{def:row-substoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}_{\geq 0}$ is row-substochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} \leq 1$ for all rows $i \in \underline{n}$.\\
    $\tT$ is strictly row-substochastic if and only if ther exists at least one row $i'$ with $\sum_{j=0}^n(\tT)_{i'j} < 1$
\end{definition}

\begin{theorem}\label{th:row-substoch_eigenvalue_0}
    If a matrix $\tT$ is strictly row-substochastic, then it has no eigenvalue $|\lambda| \geq 1$.
\end{theorem}

\begin{proof}
    Let us assume that $|\lambda| > 1$ is an eigenvalue of $\tT$. As $(\tT)_{ij}\in \RR_{\geq 0}$, using \cref{def:gershgorin_theorem}
    \[
    |\lambda-(\tT)_{ii}| = |\lambda|-(\tT)_{ii} \leq \sum_{j=1\land j\neq i}^n|(\tT)_{ij}|\]
    \[ \iff 1 < |\lambda| \leq (\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij}\ .
    \]
    \info{weil $\tT$ non negative}
    However, as $\tT$ is  strictly row-substochastic, we know that $(\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij} < 1 < |\lambda|$ for all $i \in \underline{n}$.
    Thus, any strictly row-substochastic matrix $\tT$ has no eigenvalues $|\lambda| > 1$.
\end{proof}

\begin{definition}[Jordan normal form]\label{def:jordan_normal_form}

    A discrete-time Markov chain (DTMC) is a tuple $\MC = (S, \PF,\iota_{init})$, where

Most intuitive upper and lower bounds for uEVTs and lEVTs of IDTMCs can be derived by using equation (CITE) and the upper and lower transition probability functions and initial distributions defined in \cref{def:IMC}.

\begin{theorem}[Upper/Lower Bounds for uEVTs/lEVTs]
    For any IDTMC $\IMC$, the uEVT is bounded from above by $\Initupper+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp[vt_t]$ and the lEVT is bounded from below by $\Initlower+\sum_{t\in S_{tr}}\PFlower(t,s)^\T\cdot \Exp[vt_t]$
\end{theorem}
\begin{proof}
    For any IDTMC $\IMC$ there exists a DTMC $\MC$ that satisfies $\IMC$, i.e. $\MC \models \IMC$, and $\Exp^\MC[vt_s] = \ExpUpper^\IMC[vt_s]$. As $\MC \models \IMC$ it follows that $\PF(s,s')\leq \PFupper(s,s')$ for all $s,s'\in S$.

    Thus, $\Exp^\MC[vt_s] = \iota_{init}+\sum_{t\in S_{tr}}\PF(t,s)^\T\cdot \Exp[vt_t] \leq \Initupper+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp[vt_t] = \ExpUpper^\IMC[vt_s]$.
    The identical argument can be made to prove that lower bound for the lEVT.
\end{proof}

Intuitively, the IMDP semantic allows for a probability distribution to be chosen at every step in each state within the constraints/bounds set by the IDTMC. In contrast to the UMC semantic, where each state in the satisfying DTMCs have a corresponding state in the IDTMC and thus a fixed probability distribution, while the IMDP semantic allows for multiple states in the satisfying DTMCs with different fixed probability distributions to be mapped to the same corresponding state in the IDTMC. We can thus imagine an IDTMC under the IMDP semantic as a Markov Decision Process.

\begin{definition}[Markov Descision Process (MDP)]\label{def:MDP}\cite{Bibel}\info{größtenteils übernommen}
    A \textit{Markov Decision Process} (MDP) is a tuple $D = (S,Act,A,\PF,\iota_{init})$, where
    \begin{enumerate}[label=(\roman*)]
        \item $S$ is a countable set of states
        \item Act is a set of actions
        \item $A(s)\subseteq Act$ denotes the applicable actions for all states $s\in S$ 
        \item $\PF: S\times Act\times S \rightarrow [0,1]$ is the transition probability function such that $\sum_{s'\in S}\PF(s,\alpha, s') = 1$ for all $s\in S$ and $\alpha\in A(s)$
        \item $\iota_{init}: S \rightarrow [0,1]$ is the initial distribution such that $\sum_{s\in S}\iota_{init}(s)=1$
    \end{enumerate}
\end{definition}

The most intuitive reduction from an IDTMC $\IMC$ under the IMDP semantic to a MDP $\MDP$ is to define $S^\MDP = S^\IMC$ and an uncoutable set $\cup_{s\in S} A(s) = Act$ of actions where each action $\alpha \in A(s)$ corresponds to a probability distribution \info{that is admissable} $\PFlower^\IMC(s,s')\leq \PF^\MDP(s,\alpha,s')\leq \PFupper^\IMC(s,s')$ (for $s'\in N(s)$). In other words, the actions $\alpha\in A(s)$ in a given state $s\in S$ correspond to the probability distributions that can be chosen in $s$ in any given step. However, as even the subsets $A(s)$ can be uncoutable, this reduction is trivially not "usable", as no algorithm, such as value or policy iteration, would terminate.

Instead, it is thus argued in \cite{Model-Checking_presence_Uncertainties} and \cite{Semantics_Model-Checking_omega-Regular}, that the local constraints of each state of these MDPs and IDTMCs, see in \cref{def:MDP} $(iv) - (v)$ and \cref{def:IMC} $(ii) - (v)$, can be used to limit the amount of probability distributions that need to be mapped(, as we only need to consider the most extreme "behaviour"). In the theory of linear programming, this is called a basic feasible solution (BFS) \done{verschieben zu Matheteil?}. 
In the following, we will extend the reduction from IDTMCs to MDPs (using the IMDP semantic) presented in \cite{Model-Checking_presence_Uncertainties} and \cite{Semantics_Model-Checking_omega-Regular}. We will refer to the constraints of the form $\sum_{s'\in N(s)}\PF (s,s')$, where $\PFlower(s,s')\leq\PF(s,s')\leq\PFupper(s,s')$ as $PC(s)$ (for all $s\in S_{tr}$) and those of the form $\sum_{s\in S}\iota_{init} = 1$, where $\Initlower(s) \leq \iota_{init}(s) \leq \Initupper(s)$ as $IC$ (notice not dependent on s, as only one set of inequations/constraints).

\begin{theorem}[Basic feasible solution (BFS) for $PC(s)$]
    A map $\theta^s: S\rightarrow [0,1]$ is a \textit{basic feasible solution} for the constraints $PC(s)$ if and only if $\{p_{ss'} = \theta^s(s')|s'\in S\}$ is a solution for $PC(s)$ and there exists a set $S' \subseteq S$ such that $|S'|\geq |S|-1$ and for all $s' \in S'$ either $\theta^s(s')= \PFlower(s,s')$ or $\theta^s(s')= \PFupper(s,s')$.
\end{theorem}

\begin{theorem}[BFS for $IC$]
