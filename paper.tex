\begin{definition}[Gershgorin circle theorem]\label{def:gershgorin_theorem}
    Let $\tT\in \CC^{n\times n}$ be a square complex matrix, then for every eigenvalue $\lambda$ of $\tT$ it is true that 
    \[|\lambda-(\tT)_{ii}|\leq\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|\ ,\] 
    for any $i\in \underline{n}$ and where $|\cdot|$ denotes the absolute value, i.e. magnitude.\\
    In other words, all eigenvalues of $\tT$ lie on a disk in the complex plane centered at $(\tT)_{ii}$ with radius $\sum_{j=1\land i\neq j}^n|(\tT)_{ij}|$.
\end{definition}
Matrix invertible if 1 not eigenvalue

\begin{definition}[Row-Stochastic Matrix]\label{def:row-stoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}$ is row-stochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} = 1$ for all rows $i \in \underline{n}$
\end{definition}
When considering each row as a function $f_i(j)=(\tT)_{ij}$, these functions are discrete probability distributions.

\begin{definition}[Row-Substochastic Matrix]\label{def:row-substoch_matrix}
    A square matrix $\tT \in \RR^{n\times n}$ is row-stochastic, if and only if $\sum_{j=0}^n(\tT)_{ij} \leq 1$ for all rows $i \in \underline{n}$
\end{definition}

\begin{theorem}\label{th:row-substoch_eigenvalue_0}
    If a matrix $\tT$ is row-substochastic, then it has no eigenvalue $\lambda = 1$.
\end{theorem}

\begin{proof}
    Let us assume that $1$ is an eigenvalue of $\tT$. As $(\tT)_{ij}\in \RR$, using \cref{def:gershgorin_theorem}
    \[
    |1-(\tT)_{ii}| \leq \sum_{j=1\land j\neq i}^n(\tT)_{ij} \iff 1 \leq (\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij}\ .
    \]
    However, as $\tT$ is row-substochastic, we know that $(\tT)_{ii}+\sum_{j=1\land j\neq i}^n(\tT)_{ij} < 1$.
    Thus, 1 is not an eigenvalue of $\tT$.
\end{proof}

\subsubsection{Expected Visiting Times for DTMCs}
Intuitively, the notion of EVTs describes how often we expect the model to be in a given state. To formally introduce EVTs, we fix a DTMC $\MC$ and start by introducing \textit{visiting times} along paths.
\begin{definition}[Visiting Times of Infinite Paths]
    The function $vt_s:\Path \rightarrow \RR$ describes the \textit{visiting time} of a state $s\in S$ along infinite paths in $\MC$, where for $\pi\in \Path$
    \[vt_s(\pi)=\sum_{k=0}^{\infty}\llbracket \pi[k]=s \rrbracket\]
\end{definition}

\begin{definition}[Expected Visiting Times in DTMCs]\label{def:EVTs_in_DTMCs}

\end{definition}

Thus, any DTMC whose initial distribution and transition probabilities lie in the intervals defined by the IDTMC is satisfactory.

\begin{definition}[IMDP semantic]
    An arbitrary DTMC $\MC=(T,\PF,\iota_{init})$ satisfies an IDTMC $\IMC = (S,\PFlower,\PFupper,\Initlower,\Initupper)$ if and only if there exists an injective mapping $\tau: T\rightarrow S$ such that
    \begin{enumerate}[label=(\roman*)]
        \item $\tau(t) = s\in S$ for all $t\in T \implies \Initlower(s) \leq \iota_{init}(t)\leq \Initupper(s)$
        \item $\tau(t) = s\in S$ for all $t\in T \implies \PFlower(s) \leq \PF(t)\leq \PFupper(s)$
    \end{enumerate}
\end{definition}

Most intuitive upper and lower bounds for UEVTs and LEVTs of IDTMCs can be derived by using equation (CITE) and the upper and lower transition probability functions and initial distributions defined in \cref{def:IMC}.

\begin{theorem}[Upper/Lower Bounds for UEVTs/LEVTs]
    For any IDTMC $\IMC$, the UEVT is bounded from above by $\Initupper+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp[vt_t]$ and the LEVT is bounded from below by $\Initlower+\sum_{t\in S_{tr}}\PFlower(t,s)^\T\cdot \Exp[vt_t]$
\end{theorem}
\begin{proof}
    For any IDTMC $\IMC$ there exists a DTMC $\MC$ that satisfies $\IMC$, i.e. $\MC \models \IMC$, and $\Exp^\MC[vt_s] = \ExpUpper^\IMC[vt_s]$. As $\MC \models \IMC$ it follows that $\PF(s,s')\leq \PFupper(s,s')$ for all $s,s'\in S$.

    Thus, $\Exp^\MC[vt_s] = \iota_{init}+\sum_{t\in S_{tr}}\PF(t,s)^\T\cdot \Exp[vt_t] \leq \Initupper+\sum_{t\in S_{tr}}\PFupper(t,s)^\T\cdot \Exp[vt_t] = \ExpUpper^\IMC[vt_s]$.
    The identical argument can be made to prove that lower bound for the LEVT.
\end{proof}
